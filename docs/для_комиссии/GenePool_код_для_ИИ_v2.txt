//------------------------------------------------------------------------------
// Physics kernel (stage 7): the body motion of one swimbot, working only on
// the flat arrays of SharedState, so it can run in any thread.
//
// This is the old Swimbot.updateBodyParts + updatePhysics, moved here with the
// same arithmetic in the same order (results are bit-identical; checked with
// tools/bench.cjs fingerprints).
//
//   physicsPartA: body parts move, fluid forces, swimming cost, efficiency,
//                 the constant energy drain
//   physicsPartB: starvation check, walls
// (between them the main thread may apply the climate drain)
//------------------------------------------------------------------------------
import { ONE, ONE_HALF, PI_OVER_180, ZERO } from "./MathConstants";
import { OLD_AGE_DURATION, SWIMBOT_SELECT_RADIUS_SCALAR, YOUNG_AGE_DURATION } from "./Parameters";
import { applyWaterDrag, bodyMotionFactors } from "./BodyPhysics";
import { POOL_BOTTOM, POOL_LEFT, POOL_RIGHT, POOL_TOP } from "./Pool";
import { CONTINUAL_ENERGY_DRAIN, ENERGY_EFFICIENCY_MEASUREMENT_PERIOD, ENERGY_USED_UP_SWIMMING, GENITAL_INDEX, MAX_PARTS, MOUTH_INDEX, ROOT_PART, STARVING, STARVING_TIMER_DELTA, TIMER_DELTA_INCREASE_RATE, WALL_BOUNCE } from "./SwimbotTypes";
import {
    A_X, A_Y, B_A, B_AGE, B_ANGLE, B_BASAL, B_BODY_MASS, B_COM_X, B_COM_Y, B_DEATHS, B_EFFICIENCY, B_EFFORT,
    B_ENERGY, B_ETERNAL, B_FOCUS_X, B_FOCUS_Y, B_FREQUENCY, B_GENITAL_X, B_GENITAL_Y, B_GOAL_X, B_GOAL_Y, B_GROWTH,
    B_LAST_ENERGY, B_LAST_X, B_LAST_Y, B_MOUTH_X, B_MOUTH_Y, B_NUM_PARTS, B_POS_X, B_POS_Y, B_SELECT_RADIUS,
    B_SPIN, B_SUM_LENGTHS, B_TIMER, B_TIMER_DELTA, B_VEL_X, B_VEL_Y, BOT_STRIDE, C_A, M_X, M_Y, N_X, N_Y, P_X,
    P_Y, PARAM_MAX_LIFESPAN, PART_STRIDE, PS, Q_X, Q_Y, SharedState, V_X, V_Y,
    PARAM_OB_END_RADIUS, PARAM_OB_LENGTH, PARAM_OB_MID_X, PARAM_OB_MID_Y, PARAM_OB_PERP_X, PARAM_OB_PERP_Y,
    PARAM_OBSTACLE_X1, PARAM_OBSTACLE_Y1
} from "./SharedState";

//------------------------------------------------------------------------------
// part A
//------------------------------------------------------------------------------
export function physicsPartA( s: number, st: SharedState ): void
{
    const B   = st.bots;
    const b   = s * BOT_STRIDE;
    const pd  = st.parts;
    const o   = s * PART_STRIDE;         // this swimbot's part data starts here
    const k   = s * MAX_PARTS;           // ... and its static part data
    const length    = st.partLength;
    const mass      = st.partMass;
    const angle     = st.partAngle;
    const amp       = st.partAmp;
    const phase     = st.partPhase;
    const turnAmp   = st.partTurnAmp;
    const turnPhase = st.partTurnPhase;
    const moment    = st.partMoment;
    const parentOf  = st.partParent;

    const maximumLifeSpan = st.params[ PARAM_MAX_LIFESPAN ];
    const oldAgeThreshold = maximumLifeSpan - OLD_AGE_DURATION;

    const age       = B[ b + B_AGE ];
    const eternal   = B[ b + B_ETERNAL ] !== 0;
    const numParts  = B[ b + B_NUM_PARTS ];
    const frequency = B[ b + B_FREQUENCY ];
    const sumLengths= B[ b + B_SUM_LENGTHS ];
    const bodyMass  = B[ b + B_BODY_MASS ];
    const motion    = bodyMotionFactors( bodyMass, sumLengths > ZERO ? bodyMass / sumLengths : ZERO );
    let energy      = B[ b + B_ENERGY ];
    let growth      = B[ b + B_GROWTH ];
    let timerDelta  = B[ b + B_TIMER_DELTA ];
    let posX        = B[ b + B_POS_X ];
    let posY        = B[ b + B_POS_Y ];
    let angleNow    = B[ b + B_ANGLE ];

    //==================================================================
    // update body parts
    //==================================================================
    if (( age < oldAgeThreshold ) || eternal )
    {
        growth = ( age < YOUNG_AGE_DURATION ) ? age / YOUNG_AGE_DURATION : ONE;

        // slowing down because starving, but not to a full stop
        if ( energy < STARVING )
        {
            timerDelta = energy / STARVING;
            if ( timerDelta < STARVING_TIMER_DELTA ) { timerDelta = STARVING_TIMER_DELTA; }
        }
        else
        {
            timerDelta += TIMER_DELTA_INCREASE_RATE;
            if ( timerDelta > ONE ) { timerDelta = ONE; }
        }
    }
    else
    {
        if ( age > maximumLifeSpan )
        {
            B[ b + B_DEATHS ] += 1;        // dying of old age
        }
        else
        {
            timerDelta = ONE - ( age - oldAgeThreshold ) / OLD_AGE_DURATION;   // slowing down because dying
        }
    }

    const timer = B[ b + B_TIMER ] + timerDelta * B[ b + B_EFFORT ];

    // modulators: the dot between the heading's perpendicular and the focus direction
    const radian0 = angleNow * PI_OVER_180;
    const headingX = Math.sin( radian0 );
    const headingY = Math.cos( radian0 );
    const perpX =  headingY;
    const perpY = -headingX;
    const directionDot = B[ b + B_FOCUS_X ] * perpX + B[ b + B_FOCUS_Y ] * perpY;

    // root position and angle
    let momentAdjustment = ZERO;
    for (let p = 2; p < numParts; p++) { momentAdjustment += pd[ o + p * PS + B_A ] * moment[ k + p ]; }

    pd[ o + ROOT_PART * PS + P_X ] = posX;
    pd[ o + ROOT_PART * PS + P_Y ] = posY;
    pd[ o + ROOT_PART * PS + C_A ] = angleNow - momentAdjustment;

    // parts: angle and position
    for (let p = 1; p < numParts; p++)
    {
        const i = o + p * PS;
        const j = o + parentOf[ k + p ] * PS;     // (every part from 1 on has a real parent)

        pd[ i + P_X ] = pd[ j + P_X ];
        pd[ i + P_Y ] = pd[ j + P_Y ];

        pd[ i + C_A ] = pd[ j + C_A ] + angle[ k + p ];

        if ( p > 1 ) // part 1 has nothing to bend off of
        {
            const ampModulator   = turnAmp  [ k + p ] * directionDot;
            const phaseModulator = turnPhase[ k + p ] * directionDot;
            const radian = timer * frequency + ( phase[ k + p ] + phaseModulator );
            pd[ i + B_A ] = ( amp[ k + p ] + ampModulator ) * Math.sin( radian );
            pd[ i + C_A ] += pd[ i + B_A ];
        }

        const radian = pd[ i + C_A ] * PI_OVER_180;
        let partLength = length[ k + p ];
        if ( age < YOUNG_AGE_DURATION ) { partLength *= growth; }

        const x = partLength * Math.sin( radian );
        const y = partLength * Math.cos( radian );
        pd[ i + Q_X ] = pd[ i + M_X ];
        pd[ i + Q_Y ] = pd[ i + M_Y ];
        pd[ i + M_X ] = pd[ i + P_X ];
        pd[ i + M_Y ] = pd[ i + P_Y ];
        pd[ i + P_X ] += x;
        pd[ i + P_Y ] += y;
        pd[ i + M_X ] += x * ONE_HALF;
        pd[ i + M_Y ] += y * ONE_HALF;

        pd[ i + A_X ] = pd[ i + P_X ] - pd[ j + P_X ];
        pd[ i + A_Y ] = pd[ i + P_Y ] - pd[ j + P_Y ];
        pd[ i + N_X ] =  pd[ i + A_Y ] / partLength;
        pd[ i + N_Y ] = -pd[ i + A_X ] / partLength;

        pd[ i + V_X ] = pd[ i + M_X ] - pd[ i + Q_X ];
        pd[ i + V_Y ] = pd[ i + M_Y ] - pd[ i + Q_Y ];
    }

    // center of mass, shift the body so it stays in place, center of mass again
    let comX = ZERO, comY = ZERO;
    for (let p = 1; p < numParts; p++)
    {
        comX += pd[ o + p * PS + M_X ] * mass[ k + p ];
        comY += pd[ o + p * PS + M_Y ] * mass[ k + p ];
    }
    let scale = ONE / bodyMass;
    comX *= scale;
    comY *= scale;

    const offsetX = posX - comX;
    const offsetY = posY - comY;
    for (let p = 0; p < numParts; p++)
    {
        const i = o + p * PS;
        pd[ i + P_X ] += offsetX;
        pd[ i + P_Y ] += offsetY;
        pd[ i + M_X ] += offsetX;
        pd[ i + M_Y ] += offsetY;
    }
    B[ b + B_MOUTH_X   ] = pd[ o + MOUTH_INDEX   * PS + P_X ];
    B[ b + B_MOUTH_Y   ] = pd[ o + MOUTH_INDEX   * PS + P_Y ];
    B[ b + B_GENITAL_X ] = pd[ o + GENITAL_INDEX * PS + P_X ];
    B[ b + B_GENITAL_Y ] = pd[ o + GENITAL_INDEX * PS + P_Y ];

    comX = ZERO; comY = ZERO;
    for (let p = 1; p < numParts; p++)
    {
        comX += pd[ o + p * PS + M_X ] * mass[ k + p ];
        comY += pd[ o + p * PS + M_Y ] * mass[ k + p ];
    }
    comX *= scale;
    comY *= scale;
    B[ b + B_COM_X ] = comX;
    B[ b + B_COM_Y ] = comY;

    // select radius (now and then)
    if ( age % 20 === 0 )
    {
        let selectRadius = B[ b + B_SELECT_RADIUS ];
        for (let p = 1; p < numParts; p++)
        {
            for (let q = 1; q < numParts; q++)
            {
                if ( q != p )
                {
                    const xx = pd[ o + p * PS + P_X ] - pd[ o + q * PS + P_X ];
                    const yy = pd[ o + p * PS + P_Y ] - pd[ o + q * PS + P_Y ];
                    let distance = Math.sqrt( xx * xx + yy * yy );
                    distance = SWIMBOT_SELECT_RADIUS_SCALAR * Math.sqrt( distance );
                    if ( distance > selectRadius ) { selectRadius = distance; }
                }
            }
        }
        B[ b + B_SELECT_RADIUS ] = selectRadius;
    }

    //==================================================================
    // fluid forces: the moving parts push the body forward and turn it
    //==================================================================
    let accelerationX = ZERO, accelerationY = ZERO, torque = ZERO;

    for (let p = 1; p < numParts; p++)
    {
        const i = o + p * PS;
        const fractionOfWhole = length[ k + p ] / sumLengths;

        pd[ i + V_X ] = pd[ i + M_X ] - pd[ i + Q_X ];
        pd[ i + V_Y ] = pd[ i + M_Y ] - pd[ i + Q_Y ];

        const strokeAmplitude = ( pd[ i + V_X ] * pd[ i + N_X ] + pd[ i + V_Y ] * pd[ i + N_Y ] ) * fractionOfWhole;
        const strokeForceX = pd[ i + N_X ] * strokeAmplitude;
        const strokeForceY = pd[ i + N_Y ] * strokeAmplitude;

        if ( motion.enabled )
            energy -= Math.abs( strokeAmplitude ) * ENERGY_USED_UP_SWIMMING * motion.strokeCost;
        else
            energy -= Math.abs( strokeAmplitude ) * ENERGY_USED_UP_SWIMMING;
        if ( energy < ZERO ) { energy = ZERO; }

        const partVectorFromCenterX = pd[ i + M_X ] - posX;
        const partVectorFromCenterY = pd[ i + M_Y ] - posY;
        const xx = partVectorFromCenterX * partVectorFromCenterX;
        const yy = partVectorFromCenterY * partVectorFromCenterY;
        const distance = Math.sqrt( xx * xx + yy * yy );

        if ( distance > ZERO )
        {
            if ( motion.enabled )
            {
                accelerationX += -strokeForceX * motion.acceleration;
                accelerationY += -strokeForceY * motion.acceleration;
            }
            else
            {
                accelerationX += -strokeForceX;
                accelerationY += -strokeForceY;
            }

            const partPerpendicularX =  partVectorFromCenterY;
            const partPerpendicularY = -partVectorFromCenterX;
            const perpDot = ( strokeForceX * partPerpendicularX + strokeForceY * partPerpendicularY ) / sumLengths;
            torque -= perpDot;
        }
    }

    let velX = B[ b + B_VEL_X ] + accelerationX;
    let velY = B[ b + B_VEL_Y ] + accelerationY;
    if ( motion.enabled )
    {
        const drag = applyWaterDrag( velX, velY, bodyMass, motion.drag );
        velX = drag.velocityX;
        velY = drag.velocityY;
        energy -= drag.energyCost;
        if ( energy < ZERO ) { energy = ZERO; }
    }
    const spin = B[ b + B_SPIN ] + torque;
    posX += velX;
    posY += velY;
    angleNow += spin;

    //==================================================================
    // energy efficiency (now and then), constant drain
    //==================================================================
    if ( age % ENERGY_EFFICIENCY_MEASUREMENT_PERIOD === 0 )
    {
        const dx = posX - B[ b + B_LAST_X ];
        const dy = posY - B[ b + B_LAST_Y ];
        const distanceTraveled = Math.sqrt( dx * dx + dy * dy );
        const averageSpeed = distanceTraveled / ENERGY_EFFICIENCY_MEASUREMENT_PERIOD;
        let energyLost = B[ b + B_LAST_ENERGY ] - energy;
        if ( energyLost < ZERO ) { energyLost = ZERO; }
        B[ b + B_EFFICIENCY  ] = averageSpeed / ( ONE + energyLost );
        B[ b + B_LAST_X      ] = posX;
        B[ b + B_LAST_Y      ] = posY;
        B[ b + B_LAST_ENERGY ] = energy;
    }

    energy -= CONTINUAL_ENERGY_DRAIN + B[ b + B_BASAL ];

    B[ b + B_GROWTH      ] = growth;
    B[ b + B_TIMER_DELTA ] = timerDelta;
    B[ b + B_TIMER       ] = timer;
    B[ b + B_ENERGY      ] = energy;
    B[ b + B_VEL_X       ] = velX;
    B[ b + B_VEL_Y       ] = velY;
    B[ b + B_SPIN        ] = spin;
    B[ b + B_POS_X       ] = posX;
    B[ b + B_POS_Y       ] = posY;
    B[ b + B_ANGLE       ] = angleNow;
}

//------------------------------------------------------------------------------
// part B: starvation, walls
//------------------------------------------------------------------------------
export function physicsPartB( s: number, st: SharedState ): void
{
    const B  = st.bots;
    const b  = s * BOT_STRIDE;
    const pd = st.parts;
    const o  = s * PART_STRIDE;
    const k  = s * MAX_PARTS;

    // when energy hits zero, that means death (eternal swimbots don't starve)
    if ( B[ b + B_ENERGY ] <= ZERO )
    {
        B[ b + B_ENERGY ] = ZERO;
        if ( B[ b + B_ETERNAL ] === 0 ) { B[ b + B_DEATHS ] += 1; }
    }

    const numParts   = B[ b + B_NUM_PARTS ];
    const sumLengths = B[ b + B_SUM_LENGTHS ];
    let posX  = B[ b + B_POS_X  ], posY  = B[ b + B_POS_Y  ];
    let velX  = B[ b + B_VEL_X  ], velY  = B[ b + B_VEL_Y  ];
    let goalX = B[ b + B_GOAL_X ], goalY = B[ b + B_GOAL_Y ];

    const normalizeGoal = () =>
    {
        const m = Math.sqrt( goalX * goalX + goalY * goalY );
        if ( m > 0 ) { goalX /= m; goalY /= m; } else { goalX = 1.0; goalY = 0.0; }
    };

    if ( posX < POOL_LEFT + sumLengths * ONE_HALF )
    {
        for (let p = 1; p < numParts; p++)
        {
            const limit = POOL_LEFT + ( st.partLength[ k + p ] + st.partWidth[ k + p ] );
            if ( pd[ o + p * PS + P_X ] < limit )
            {
                const penetration = limit - pd[ o + p * PS + P_X ];
                posX  += penetration * WALL_BOUNCE;
                velX  += penetration * WALL_BOUNCE;
                goalX += penetration * WALL_BOUNCE;
                normalizeGoal();
            }
        }
    }
    else if ( posX > POOL_RIGHT - sumLengths * ONE_HALF )
    {
        for (let p = 1; p < numParts; p++)
        {
            const limit = POOL_RIGHT - ( st.partLength[ k + p ] + st.partWidth[ k + p ] );
            if ( pd[ o + p * PS + P_X ] > limit )
            {
                const penetration = limit - pd[ o + p * PS + P_X ];
                posX  += penetration * WALL_BOUNCE;
                velX  += penetration * WALL_BOUNCE;
                goalX += penetration * WALL_BOUNCE;
                normalizeGoal();
            }
        }
    }

    if ( posY < POOL_TOP + sumLengths * ONE_HALF )
    {
        for (let p = 1; p < numParts; p++)
        {
            const limit = POOL_TOP + ( st.partLength[ k + p ] + st.partWidth[ k + p ] );
            if ( pd[ o + p * PS + P_Y ] < limit )
            {
                const penetration = limit - pd[ o + p * PS + P_Y ];
                posY  += penetration * WALL_BOUNCE;
                velY  += penetration * WALL_BOUNCE;
                goalY += penetration * WALL_BOUNCE;
                normalizeGoal();
            }
        }
    }
    else if ( posY > POOL_BOTTOM - sumLengths * ONE_HALF )
    {
        for (let p = 1; p < numParts; p++)
        {
            const limit = POOL_BOTTOM - ( st.partLength[ k + p ] + st.partWidth[ k + p ] );
            if ( pd[ o + p * PS + P_Y ] > limit )
            {
                const penetration = limit - pd[ o + p * PS + P_Y ];
                posY  += penetration * WALL_BOUNCE;
                velY  += penetration * WALL_BOUNCE;
                goalY += penetration * WALL_BOUNCE;
                normalizeGoal();
            }
        }
    }

    B[ b + B_POS_X  ] = posX;  B[ b + B_POS_Y  ] = posY;
    B[ b + B_VEL_X  ] = velX;  B[ b + B_VEL_Y  ] = velY;
    B[ b + B_GOAL_X ] = goalX; B[ b + B_GOAL_Y ] = goalY;
}

//------------------------------------------------------------------------------
// the whole physics of one swimbot for one step (phased step: the climate
// drain was already applied before)
//------------------------------------------------------------------------------
export function physicsStep( s: number, st: SharedState ): void
{
    physicsPartA( s, st );
    physicsPartB( s, st );
    obstacleCollision( s, st );
}

//------------------------------------------------------------------------------
// bumping into the obstacle pushes the swimbot away (as Obstacle.getCollision,
// with the push of GenePool: force x 1.2 added to the velocity)
//------------------------------------------------------------------------------
export function obstacleCollision( s: number, st: SharedState ): void
{
    const B = st.bots, b = s * BOT_STRIDE, P = st.params;
    const endRadius = P[ PARAM_OB_END_RADIUS ];
    if ( endRadius <= ZERO ) return;
    let radius = B[ b + B_SUM_LENGTHS ] * ONE_HALF;
    if ( radius < endRadius ) { radius = endRadius; }

    const xx = B[ b + B_POS_X ] - P[ PARAM_OB_MID_X ];
    const yy = B[ b + B_POS_Y ] - P[ PARAM_OB_MID_Y ];
    const distanceSquared = xx * xx + yy * yy;
    const ll = P[ PARAM_OB_LENGTH ] * ONE_HALF + endRadius + radius;

    if ( distanceSquared < ll * ll )
    {
        const perpX = P[ PARAM_OB_PERP_X ], perpY = P[ PARAM_OB_PERP_Y ];
        const testX = B[ b + B_POS_X ] - P[ PARAM_OBSTACLE_X1 ];
        const testY = B[ b + B_POS_Y ] - P[ PARAM_OBSTACLE_Y1 ];
        const dot = testX * perpX + testY * perpY;

        if ( Math.abs( dot ) < radius )
        {
            let penetration = ( ONE - ( dot / radius ) );
            if ( dot < ZERO ) { penetration *= -ONE; }
            B[ b + B_VEL_X ] += ( perpX * penetration ) * 1.2;
            B[ b + B_VEL_Y ] += ( perpY * penetration ) * 1.2;
        }
    }
}

=================== BodyPhysics.ts ===================
import { BODY_HYDRODYNAMICS_ENABLED } from "./Parameters";

const REFERENCE_BODY_MASS = 450;
const REFERENCE_BODY_WIDTH = 3.5;
const WATER_DRAG_COEFFICIENT = 0.004;
const WATER_DRAG_ENERGY_SCALE = 0.0001;
const MAX_WATER_DRAG_FRACTION = 0.30;

export interface BodyMotionFactors
{
    enabled: boolean;
    acceleration: number;
    drag: number;
    strokeCost: number;
}

function clamp( value: number, min: number, max: number ): number
{
    return Math.max( min, Math.min( max, value ));
}

// Mass is the current segment-area proxy (sum of length * width). Width is its
// length-weighted mean. The aspect factor distinguishes broad shapes from
// slimmer bodies without assigning a preferred climate or behaviour.
export function bodyMotionFactors( mass: number, meanWidth: number ): BodyMotionFactors
{
    if ( !BODY_HYDRODYNAMICS_ENABLED || !( mass > 0 ) || !( meanWidth > 0 ))
    {
        return { enabled: false, acceleration: 1, drag: 1, strokeCost: 1 };
    }

    const relativeMass = clamp( mass / REFERENCE_BODY_MASS, 0.25, 4 );
    const relativeBreadth = clamp(
        ( meanWidth / REFERENCE_BODY_WIDTH ) / Math.sqrt( relativeMass ),
        0.5,
        2
    );
    const massScale = Math.pow( relativeMass, -1 / 3 );
    return {
        enabled: true,
        acceleration: clamp( massScale * Math.sqrt( relativeBreadth ), 0.4, 2.2 ),
        // (drag: by the breadth only - a narrow big body cruises cheaply, a
        //  broad one is a slow "tank"; it was also x mass^-1/3, which left big
        //  bodies only the costs)
        drag: clamp( relativeBreadth * relativeBreadth, 0.2, 4 ),
        strokeCost: clamp( Math.pow( relativeMass, 1 / 3 ) * Math.sqrt( relativeBreadth ), 0.4, 2.2 ),
    };
}

export interface WaterDragResult
{
    velocityX: number;
    velocityY: number;
    energyCost: number;
}

// Quadratic drag damps fast movement more than slow movement. Its energy cost
// is estimated from the kinetic energy lost during this step, then converted
// to the simulation's energy units.
export function applyWaterDrag( velocityX: number, velocityY: number, mass: number, dragScale: number ): WaterDragResult
{
    if ( !BODY_HYDRODYNAMICS_ENABLED || !( mass > 0 ) || !( dragScale > 0 ))
    {
        return { velocityX, velocityY, energyCost: 0 };
    }

    const speed = Math.sqrt( velocityX * velocityX + velocityY * velocityY );
    if ( !( speed > 0 )) return { velocityX, velocityY, energyCost: 0 };

    const fraction = Math.min( MAX_WATER_DRAG_FRACTION, WATER_DRAG_COEFFICIENT * dragScale * speed );
    const retained = 1 - fraction;
    const kineticEnergyLost = 0.5 * mass * speed * speed * ( 1 - retained * retained );
    return {
        velocityX: velocityX * retained,
        velocityY: velocityY * retained,
        energyCost: kineticEnergyLost * WATER_DRAG_ENERGY_SCALE,
    };
}

=================== ThermalPhysics.ts ===================
const REFERENCE_BODY_MASS = 450;
const REFERENCE_BODY_WIDTH = 3.5;
const REFERENCE_EXCHANGE_RATE_PER_STEP = 0.002;

function clamp( value: number, min: number, max: number ): number
{
    return Math.max( min, Math.min( max, value ));
}

// A simple Newton-style heat exchange model for this normalized, 2D world.
// Mass and width change how quickly internal temperature approaches the local
// felt-temperature signal; this function does not produce heat on its own.
export function advanceBodyTemperature( bodyTemperature: number, ambientTemperature: number, bodyMass: number, meanBodyWidth: number ): number
{
    if ( !Number.isFinite( ambientTemperature ) ) return 0.5;
    if ( !Number.isFinite( bodyTemperature ) ) return clamp( ambientTemperature, 0, 1 );
    if ( !( bodyMass > 0 ) || !( meanBodyWidth > 0 ) ) return clamp( ambientTemperature, 0, 1 );

    // In this 2D simulation, mass × mean width is a rough 3D volume proxy.
    const relativeVolume = clamp(
        ( bodyMass / REFERENCE_BODY_MASS ) * ( meanBodyWidth / REFERENCE_BODY_WIDTH ),
        0.25,
        4
    );
    const exchangeRate = REFERENCE_EXCHANGE_RATE_PER_STEP * Math.pow( relativeVolume, -1 / 3 );
    const exchangeFraction = 1 - Math.exp( -exchangeRate );
    return clamp( bodyTemperature + ( ambientTemperature - bodyTemperature ) * exchangeFraction, 0, 1 );
}

=================== DecisionKernel.ts ===================
//------------------------------------------------------------------------------
// Decision kernel (stage 7, part 3): what a swimbot's brain does in a step,
// working only on the shared arrays (SharedState) - so it can run in any
// thread, like the physics (PhysicsKernel.ts) and the looking around
// (PerceptionKernel.ts).
//
//   decideStep   - the start of a step: age, the neural brain thinking, the
//                  brain state, "am I close enough to eat / bite / mate",
//                  which way to swim, and the climate's energy drain

const PI_OVER_180 = Math.PI / 180;

//------------------------------------------------------------------------------
// each swimbot's own random numbers (mulberry32)
//------------------------------------------------------------------------------
export function seedRandom( st: SharedState, s: number, seed: number ): void { st.random[s] = seed >>> 0; }

export function botRandom( st: SharedState, s: number ): number
{
    let t = ( st.random[s] = ( st.random[s] + 0x6D2B79F5 ) >>> 0 );
    t = Math.imul( t ^ ( t >>> 15 ), t | 1 );
    t ^= t + Math.imul( t ^ ( t >>> 7 ), t | 61 );
    return (( t ^ ( t >>> 14 )) >>> 0 ) / 4294967296;
}

//------------------------------------------------------------------------------
// the environment, from the shared copies (as Environment.temperatureAt)
//------------------------------------------------------------------------------
export function temperatureAt( st: SharedState, x: number, y: number ): number
{
    const P = st.params;
    if ( P[ SS.PARAM_ENV_FIELDS ] !== 1 ) return 0.5;

    const fx = ( x - POOL_LEFT ) / ( POOL_RIGHT  - POOL_LEFT ) * FIELD_SIZE - 0.5;
    const fy = ( y - POOL_TOP  ) / ( POOL_BOTTOM - POOL_TOP  ) * FIELD_SIZE - 0.5;
    let i = Math.floor( fx ), j = Math.floor( fy );
    const ax = fx - i, ay = fy - j;
    const i0 = Math.min( FIELD_SIZE - 1, Math.max( 0, i     ));
    const i1 = Math.min( FIELD_SIZE - 1, Math.max( 0, i + 1 ));
    const j0 = Math.min( FIELD_SIZE - 1, Math.max( 0, j     ));
    const j1 = Math.min( FIELD_SIZE - 1, Math.max( 0, j + 1 ));
    const f = st.temperature;
    const base = ( f[ j0 * FIELD_SIZE + i0 ] * ( 1 - ax ) + f[ j0 * FIELD_SIZE + i1 ] * ax ) * ( 1 - ay )
               + ( f[ j1 * FIELD_SIZE + i0 ] * ( 1 - ax ) + f[ j1 * FIELD_SIZE + i1 ] * ax ) * ay;

    const latitude  = Math.abs( (( y - POOL_TOP ) / ( POOL_BOTTOM - POOL_TOP )) * 2 - 1 );
    const amplitude = ( P[ SS.PARAM_PETRI ] === 1 ) ? 0 : SEASON_AMPLITUDE_EQUATOR + ( SEASON_AMPLITUDE_POLES - SEASON_AMPLITUDE_EQUATOR ) * latitude;
    const t = base + amplitude * Math.sin( 2 * Math.PI * P[ SS.PARAM_TIME ] / SEASON_PERIOD );
    return t < 0 ? 0 : ( t > 1 ? 1 : t );
}

function seasonPhase( st: SharedState ): number { return Math.sin( 2 * Math.PI * st.params[ SS.PARAM_TIME ] / SEASON_PERIOD ); }

function plantCell( x: number, y: number ): number
{
    const c = Math.min( SS.PLANT_COLUMNS - 1, Math.max( 0, Math.floor( ( x - POOL_LEFT ) / PLANT_CELL_SIZE )));
    const r = Math.min( SS.PLANT_ROWS    - 1, Math.max( 0, Math.floor( ( y - POOL_TOP  ) / PLANT_CELL_SIZE )));
    return r * SS.PLANT_COLUMNS + c;
}

// the temperature a swimbot feels at a point (dark bodies feel warmer)
function feltTemperature( st: SharedState, s: number, x: number, y: number ): number
{
    return temperatureAt( st, x, y ) + THERMOMELANISM * ( 1 - 2 * st.brightness[s] );
}

function bodyTemperature( st: SharedState, s: number ): number
{
    const M = st.mind, m = s * SS.MIND_STRIDE, B = st.bots, b = s * SS.BOT_STRIDE;
    return ( BODY_TEMPERATURE_INERTIA_ENABLED && M[ m + SS.M_BODY_TEMP_SET ] === 1 )
        ? M[ m + SS.M_BODY_TEMP ]
        : feltTemperature( st, s, B[ b + SS.B_POS_X ], B[ b + SS.B_POS_Y ] );
}

function updateThermalFit( st: SharedState, s: number ): void
{
    const M = st.mind, m = s * SS.MIND_STRIDE;
    M[ m + SS.M_THERMAL_FIT ] = ( st.params[ SS.PARAM_ENV_ACTIVE ] === 1 )
        ? thermalFit( bodyTemperature( st, s ), M[ m + SS.M_THERMAL_OPT ], M[ m + SS.M_THERMAL_WIDTH ] )
        : 1;
}

function updateBodyTemperature( st: SharedState, s: number ): void
{
    const M = st.mind, m = s * SS.MIND_STRIDE, B = st.bots, b = s * SS.BOT_STRIDE;
    const ambient = feltTemperature( st, s, B[ b + SS.B_POS_X ], B[ b + SS.B_POS_Y ] );
    const sumLengths = B[ b + SS.B_SUM_LENGTHS ];
    const meanBodyWidth = sumLengths > 0 ? st.mass[s] / sumLengths : 0;
    if ( !BODY_TEMPERATURE_INERTIA_ENABLED )
    {
        M[ m + SS.M_BODY_TEMP ] = ambient;
        M[ m + SS.M_BODY_TEMP_SET ] = 0;
        return;
    }
    if ( M[ m + SS.M_BODY_TEMP_SET ] !== 1 )
    {
        M[ m + SS.M_BODY_TEMP ] = advanceBodyTemperature( ambient, ambient, st.mass[s], meanBodyWidth );
        M[ m + SS.M_BODY_TEMP_SET ] = 1;
    }
    else
    {
        M[ m + SS.M_BODY_TEMP ] = advanceBodyTemperature( M[ m + SS.M_BODY_TEMP ], ambient, st.mass[s], meanBodyWidth );
    }
}

const _touchBuffer = new Int32Array( 4096 );
const EAT_RADIUS = SWIMBOT_MOUTH_LENGTH + FOOD_BIT_GRAB_RADIUS;

// a food bit this swimbot can eat, touching its mouth (-1: none)
function edibleFoodAtMouth( st: SharedState, s: number, mouthX: number, mouthY: number ): number
{
    const meatEater = isPredator( st, s );
    const n = st.foodGrid.collect( mouthX, mouthY, EAT_RADIUS, _touchBuffer );
    for (let i = 0; i < n; i++)
    {
        const f = _touchBuffer[i];
        if ( st.liveFood[f] !== 1 ) continue;
        if (( st.foodType[f] === 2 ) !== meatEater ) continue;
        const dx = st.foodX[f] - mouthX, dy = st.foodY[f] - mouthY;
        if ( dx * dx + dy * dy < EAT_RADIUS * EAT_RADIUS ) return f;
    }
    return -1;
}

// is the mouth at one of the body parts of swimbot p? (part ends and middles,
// within the mouth length plus the part's width)
function mouthTouchesBody( st: SharedState, p: number, mouthX: number, mouthY: number ): boolean
{
    const numParts = st.bots[ p * SS.BOT_STRIDE + SS.B_NUM_PARTS ];
    const o = p * SS.PART_STRIDE, w = p * MAX_PARTS;
    for (let k = 0; k < numParts; k++)
    {
        const i = o + k * SS.PS;
        const reach = SWIMBOT_MOUTH_LENGTH + st.partWidth[ w + k ];
        const ex = st.parts[ i + SS.P_X ] - mouthX, ey = st.parts[ i + SS.P_Y ] - mouthY;
        if ( ex * ex + ey * ey < reach * reach ) return true;
        const mx = st.parts[ i + SS.M_X ] - mouthX, my = st.parts[ i + SS.M_Y ] - mouthY;
        if ( mx * mx + my * my < reach * reach ) return true;
    }
    return false;
}

function isPredator( st: SharedState, s: number ): boolean { return st.predator[s] > PREDATOR_LEVEL_THRESHOLD; }

function foodIsMeatAndAlive( st: SharedState, f: number ): boolean
{
    return ( f >= 0 ) && ( st.liveFood[f] === 1 ) && ( st.foodType[f] === 2 );
}

function normalize( B: Float64Array, k: number ): void
{
    const x = B[k], y = B[k + 1];
    const length = Math.sqrt( x * x + y * y );
    if ( length > 0 ) { B[k] = x / length; B[k + 1] = y / length; }
    else              { B[k] = 1;          B[k + 1] = 0;          }
}

// the direction to the goal = from (fromX, fromY) to (toX, toY), normalized
function aim( B: Float64Array, b: number, toX: number, toY: number, fromX: number, fromY: number ): void
{
    B[ b + SS.B_GOAL_X ] = toX - fromX;
    B[ b + SS.B_GOAL_Y ] = toY - fromY;
    normalize( B, b + SS.B_GOAL_X );
}

//------------------------------------------------------------------------------
// kinesis ("run and tumble"): see Swimbot history; compare how good it is
// here with how good it was, and maybe turn
//------------------------------------------------------------------------------
function kinesis( st: SharedState, s: number ): void
{
    const M = st.mind, m = s * SS.MIND_STRIDE, B = st.bots, b = s * SS.BOT_STRIDE;
    updateThermalFit( st, s );

    const px = B[ b + SS.B_POS_X ], py = B[ b + SS.B_POS_Y ];
    const fit = M[ m + SS.M_THERMAL_FIT ];

    if ( fit < EMERGENCY_THERMAL_FIT )
    {
        const gx = B[ b + SS.B_GOAL_X ], gy = B[ b + SS.B_GOAL_Y ];
        const length = Math.sqrt( gx * gx + gy * gy );
        let dx = ( length > 0 ) ? gx / length : 1;
        let dy = ( length > 0 ) ? gy / length : 0;
        let bestFit = -1, bestX = dx, bestY = dy;
        for (let k = 0; k < 4; k++)
        {
            const f = thermalFit( feltTemperature( st, s, px + dx * EMERGENCY_PROBE_DISTANCE, py + dy * EMERGENCY_PROBE_DISTANCE ),
                                  M[ m + SS.M_THERMAL_OPT ], M[ m + SS.M_THERMAL_WIDTH ] );
            if ( f > bestFit ) { bestFit = f; bestX = dx; bestY = dy; }
            const t = dx; dx = -dy; dy = t;   // turn 90 degrees
        }
        B[ b + SS.B_GOAL_X ] = bestX;
        B[ b + SS.B_GOAL_Y ] = bestY;
        M[ m + SS.M_COMFORT ] = -1;
        return;
    }

    const cell = plantCell( px, py );
    const food = ( st.plantWarm[ cell ] * st.effWarm[s] + st.plantCold[ cell ] * st.effCold[s] ) / 6;
    const comfort = ( 0.05 + fit * M[ m + SS.M_THERMAL_PEAK ] ) * ( 0.05 + food );
    const previous = M[ m + SS.M_COMFORT ];

    M[ m + SS.M_COMFORT_TREND ] = ( previous < 0 ) ? 0 : ( comfort > previous ? 1 : ( comfort < previous ? -1 : 0 ));

    if (( previous >= 0 ) && ( comfort < previous ) && ( botRandom( st, s ) < KINESIS_TURN_CHANCE ))
    {
        const turn = ( -1 + 2 * botRandom( st, s )) * KINESIS_MAX_TURN * PI_OVER_180;
        const c = Math.cos( turn ), sn = Math.sin( turn );
        const x = B[ b + SS.B_GOAL_X ], y = B[ b + SS.B_GOAL_Y ];
        B[ b + SS.B_GOAL_X ] = x * c - y * sn;
        B[ b + SS.B_GOAL_Y ] = x * sn + y * c;
    }

    M[ m + SS.M_COMFORT ] = comfort;
}

function wanderFocus( st: SharedState, s: number ): void
{
    const B = st.bots, b = s * SS.BOT_STRIDE;
    if (( st.params[ SS.PARAM_ENV_ACTIVE ] === 1 ) && ((( B[ b + SS.B_AGE ] + st.mind[ s * SS.MIND_STRIDE + SS.M_INDEX ] ) % KINESIS_PERIOD ) === 0 ))
    {
        kinesis( st, s );
    }

    let gx = B[ b + SS.B_GOAL_X ], gy = B[ b + SS.B_GOAL_Y ];
    let length = Math.sqrt( gx * gx + gy * gy );
    if ( length === 0 )
    {
        gx = -0.5 + botRandom( st, s );
        gy = -0.5 + botRandom( st, s );
        length = Math.sqrt( gx * gx + gy * gy );
    }
    gx += ( -BRAIN_WANDER_AMOUNT * 0.5 + botRandom( st, s ) * BRAIN_WANDER_AMOUNT );
    gy += ( -BRAIN_WANDER_AMOUNT * 0.5 + botRandom( st, s ) * BRAIN_WANDER_AMOUNT );
    B[ b + SS.B_GOAL_X ] = gx / length;
    B[ b + SS.B_GOAL_Y ] = gy / length;
}

//------------------------------------------------------------------------------
// one thought of the neural brain (was Swimbot.think)
//------------------------------------------------------------------------------
const _newHidden = new Float64Array( NUM_HIDDEN );

export function neuralThink( st: SharedState, s: number ): void
{
    const M = st.mind, m = s * SS.MIND_STRIDE, B = st.bots, b = s * SS.BOT_STRIDE;
    const energy    = B[ b + SS.B_ENERGY ];
    const hunger    = M[ m + SS.M_HUNGER ];
    const predator  = M[ m + SS.M_PRED_LEVEL ] > PREDATOR_LEVEL_THRESHOLD;
    const nearPred  = M[ m + SS.M_NEAR_PRED ], calmPred = M[ m + SS.M_CALM_PRED ];
    const threat    = ( nearPred >= 0 ) || ( calmPred >= 0 );
    const xi = s * NUM_INPUTS, x = st.nnInputs;

    x[ xi + IN_BIAS     ] = 1;
    x[ xi + IN_HUNGER   ] = Math.max( -1, Math.min( 1, ( hunger - energy ) / Math.max( 1, hunger )));
    x[ xi + IN_ENERGY   ] = Math.min( 2, energy / 100 );
    x[ xi + IN_AGE      ] = B[ b + SS.B_AGE ] / st.params[ SS.PARAM_MAX_LIFESPAN ];
    x[ xi + IN_FIT      ] = M[ m + SS.M_THERMAL_FIT ];
    x[ xi + IN_TREND    ] = M[ m + SS.M_COMFORT_TREND ];
    x[ xi + IN_FOOD     ] = M[ m + SS.M_SEEN_FOOD ];
    x[ xi + IN_MATE     ] = ( M[ m + SS.M_NUM_SEEN ] > 0 ) ? 1 : 0;
    x[ xi + IN_THREAT   ] = M[ m + SS.M_THREAT_CLOSE ];
    x[ xi + IN_PREDATOR_NEAR ] = M[ m + SS.M_PRED_CLOSE ];
    x[ xi + IN_PREY     ] = M[ m + SS.M_SEEN_PREY ];
    x[ xi + IN_CROWD    ] = Math.min( 1, M[ m + SS.M_NUM_SEEN ] / 10 );
    x[ xi + IN_PREDATOR ] = predator ? 1 : 0;

    // directions, as the swimbot sees them (heading as in the physics)
    const radian   = B[ b + SS.B_ANGLE ] * PI_OVER_180;
    const headingX = Math.sin( radian ), headingY = Math.cos( radian );
    const px = B[ b + SS.B_POS_X ], py = B[ b + SS.B_POS_Y ];
    const toward = ( tx: number, ty: number, sideIndex: number, aheadIndex: number ) =>
    {
        let dx = tx - px, dy = ty - py;
        const length = Math.sqrt( dx * dx + dy * dy );
        if ( length <= 0 ) { x[ xi + sideIndex ] = 0; x[ xi + aheadIndex ] = 0; return; }
        dx /= length; dy /= length;
        x[ xi + aheadIndex ] = dx * headingX + dy * headingY;
        x[ xi + sideIndex  ] = dx * headingY - dy * headingX;
    };
    x[ xi + IN_FOOD_SIDE ] = x[ xi + IN_FOOD_AHEAD ] = x[ xi + IN_NEAR_SIDE ] = x[ xi + IN_NEAR_AHEAD ] = 0;
    x[ xi + IN_THREAT_SIDE ] = x[ xi + IN_THREAT_AHEAD ] = x[ xi + IN_KIN ] = 0;
    if ( M[ m + SS.M_SEEN_FOOD ] === 1 ) { toward( M[ m + SS.M_SEEN_FOOD_X ], M[ m + SS.M_SEEN_FOOD_Y ], IN_FOOD_SIDE, IN_FOOD_AHEAD ); }
    const nearest = M[ m + SS.M_NEAREST_SEEN ];
    if ( nearest >= 0 )
    {
        toward( st.botX[ nearest ], st.botY[ nearest ], IN_NEAR_SIDE, IN_NEAR_AHEAD );
        const difference = ( Math.abs( st.colorR[ nearest ] - st.colorR[s] ) + Math.abs( st.colorG[ nearest ] - st.colorG[s] ) + Math.abs( st.colorB[ nearest ] - st.colorB[s] )) / 3;
        x[ xi + IN_KIN ] = 1 - difference;
    }
    if ( threat )
    {
        const p = (( nearPred >= 0 ) ? nearPred : calmPred ) * SS.BOT_STRIDE;
        toward( B[ p + SS.B_POS_X ], B[ p + SS.B_POS_Y ], IN_THREAT_SIDE, IN_THREAT_AHEAD );
    }
    x[ xi + IN_SEASON ] = ( st.params[ SS.PARAM_ENV_ACTIVE ] === 1 ) ? seasonPhase( st ) : 0;

    const w = st.brainWeights.subarray( s * BRAIN_SIZE, ( s + 1 ) * BRAIN_SIZE );
    const inputs  = st.nnInputs.subarray( xi, xi + NUM_INPUTS );
    const hidden  = st.nnHidden.subarray( s * NUM_HIDDEN, ( s + 1 ) * NUM_HIDDEN );
    const outputs = st.nnOutputs.subarray( s * NUM_OUTPUTS, ( s + 1 ) * NUM_OUTPUTS );
    think( w, inputs, hidden, _newHidden, outputs );

    //------------------------------------------------------
    // the strongest possible drive wins (the current goal
    // gets a small bonus so the swimbot doesn't dither)
    //------------------------------------------------------
    const current = M[ m + SS.M_GOAL ];
    let best = OUT_MATE, bestValue = -Infinity;
    for ( const d of DRIVE_OUTPUTS )
    {
        if (( d === OUT_SELF_EAT ) && !canSelfFeed( st, s )) continue;
        if (( d === OUT_FLEE ) && ( !threat || predator )) continue;   // nothing to flee from
        if (( d === OUT_HUNT ) && !predator ) continue;                // only predators hunt
        if (( d === OUT_FOOD ) && predator ) continue;                 // predators don't eat plants
        const value = outputs[d] + ( d === current ? DRIVE_HYSTERESIS : 0 );
        if ( value > bestValue ) { bestValue = value; best = d; }
    }
    M[ m + SS.M_GOAL ] = best;

    //------------------------------------------------------
    // how to move
    //------------------------------------------------------
    const envActive = st.params[ SS.PARAM_ENV_ACTIVE ] === 1;
    B[ b + SS.B_EFFORT ]      = EFFORT_MIN + EFFORT_RANGE * sigmoid( outputs[ OUT_EFFORT ] );
    M[ m + SS.M_STEER ]       = Math.tanh( outputs[ OUT_STEER ] ) * STEER_MAX_DEGREES * PI_OVER_180;
    M[ m + SS.M_SOCIAL_W ]    = Math.tanh( outputs[ OUT_SOCIAL ] );
    const climateWeight       = envActive ? sigmoid( outputs[ OUT_CLIMATE ] ) : 0;
    M[ m + SS.M_CLIMATE_W ]   = climateWeight;

    if ( climateWeight > 0.05 )
    {
        // which way does the temperature suit me better?
        const d = CLIMATE_PROBE_DISTANCE, opt = M[ m + SS.M_THERMAL_OPT ], width = M[ m + SS.M_THERMAL_WIDTH ];
        const gx = thermalFit( feltTemperature( st, s, px + d, py ), opt, width ) - thermalFit( feltTemperature( st, s, px - d, py ), opt, width );
        const gy = thermalFit( feltTemperature( st, s, px, py + d ), opt, width ) - thermalFit( feltTemperature( st, s, px, py - d ), opt, width );
        const length = Math.sqrt( gx * gx + gy * gy );
        M[ m + SS.M_CLIMATE_X ] = ( length > 1e-4 ) ? gx / length : 0;
        M[ m + SS.M_CLIMATE_Y ] = ( length > 1e-4 ) ? gy / length : 0;
    }
    else
    {
        M[ m + SS.M_CLIMATE_X ] = M[ m + SS.M_CLIMATE_Y ] = 0;
    }
}

function canSelfFeed( st: SharedState, s: number ): boolean
{
    const M = st.mind, m = s * SS.MIND_STRIDE;
    return SELF_CANNIBALISM_ENABLED && ( st.params[ SS.PARAM_ENV_ACTIVE ] === 1 ) && ( M[ m + SS.M_PARTS_LOST ] !== 1 )
        && ( M[ m + SS.M_PHENO_PARTS ] > 4 ) && ( st.bots[ s * SS.BOT_STRIDE + SS.B_ENERGY ] < M[ m + SS.M_MAX_ENERGY ] );
}

//------------------------------------------------------------------------------
// the start of a step (was Swimbot.decide + writeTargetChoiceState +
// applyClimateDrain)
//------------------------------------------------------------------------------
export function decideStep( st: SharedState, s: number, counters: number ): void
{
    const M = st.mind, m = s * SS.MIND_STRIDE, B = st.bots, b = s * SS.BOT_STRIDE;
    const random = () => botRandom( st, s );

    M[ m + SS.M_TRY_EAT  ] = 0;
    M[ m + SS.M_TRY_HUNT ] = 0;
    B[ b + SS.B_DEATHS   ] = 0;

    const f = M[ m + SS.M_FOOD ], mate = M[ m + SS.M_MATE ];
    const foodAlive = ( f >= 0 ) && ( st.liveFood[f] === 1 );
    const mateAlive = ( mate >= 0 ) && ( st.liveSwimbot[ mate ] === 1 );

    // age
    const age = ++ B[ b + SS.B_AGE ];
    const index = M[ m + SS.M_INDEX ];

    // a dropped part grows back after a while, if I can pay for it (main thread)
    if ( M[ m + SS.M_PARTS_LOST ] === 1 )
    {
        M[ m + SS.M_REGROW ] --;
        if (( M[ m + SS.M_REGROW ] <= 0 ) && ( B[ b + SS.B_ENERGY ] > M[ m + SS.M_HUNGER ] + M[ m + SS.M_AUTOTOMY_LOSS ] ))
        {
            M[ m + SS.M_EVENTS ] |= SS.EVENT_REGROW;
        }
    }

    // a predator's search image fades
    if ( st.imageStrength[s] > 0 ) { st.imageStrength[s] *= SEARCH_IMAGE_FORGETTING; }

    if ( M[ m + SS.M_HUNT_CD ] > 0 ) { M[ m + SS.M_HUNT_CD ] --; }

    if ( age % BRAIN_SENSORY_UPDATE_PERIOD == 0 ) { M[ m + SS.M_READY ] = 1; }

    // neural brain: think now and then
    if (( M[ m + SS.M_NEURAL ] === 1 ) && ((( age + index ) % THINK_PERIOD ) === 0 )) { neuralThink( st, s ); }

    // brain
    M[ m + SS.M_BRAIN_ENERGY ] = B[ b + SS.B_ENERGY ];
    brainUpdate( M, m, random );
    const state = M[ m + SS.M_STATE ];

    if ( M[ m + SS.M_SELF_FEED_CD ] > 0 ) M[ m + SS.M_SELF_FEED_CD ] --;
    if (( state === BRAIN_STATE_SELF_FEED ) && ( M[ m + SS.M_SELF_FEED_CD ] === 0 ))
    {
        M[ m + SS.M_EVENTS ] |= SS.EVENT_SELF_FEED;
        M[ m + SS.M_SELF_FEED_CD ] = THINK_PERIOD;
    }

    if (( state === BRAIN_STATE_PURSUING_PREY ) && foodIsMeatAndAlive( st, f ))
    {
        if ( f !== M[ m + SS.M_LAST_MEAT ] ) st.counters[ counters + SS.K_MEAT_TARGETS_PURSUING ] ++;
        M[ m + SS.M_LAST_MEAT ] = f;
    }
    else if ( state !== BRAIN_STATE_PURSUING_PREY ) { M[ m + SS.M_LAST_MEAT ] = -1; }

    const mouthX = B[ b + SS.B_MOUTH_X ], mouthY = B[ b + SS.B_MOUTH_Y ];
    const prey = M[ m + SS.M_PREY ];
    const preyValid = ( prey >= 0 ) && ( st.liveSwimbot[ prey ] === 1 );

    //-------------------------------------------------
    // close enough to eat, mate or bite?
    //-------------------------------------------------
    if ( state === BRAIN_STATE_PURSUING_FOOD )
    {
        if ( foodAlive )
        {
            const xx = st.foodX[f] - mouthX, yy = st.foodY[f] - mouthY;
            if ( Math.sqrt( xx * xx + yy * yy ) < SWIMBOT_MOUTH_LENGTH + FOOD_BIT_GRAB_RADIUS ) { M[ m + SS.M_TRY_EAT ] = 1; }
        }
    }
    else if ( state === BRAIN_STATE_PURSUING_MATE )
    {
        if ( mateAlive )
        {
            const xx = st.genitalX[ mate ] - B[ b + SS.B_GENITAL_X ], yy = st.genitalY[ mate ] - B[ b + SS.B_GENITAL_Y ];
            if ( Math.sqrt( xx * xx + yy * yy ) < SWIMBOT_GENITAL_LENGTH ) { M[ m + SS.M_TRY_MATE ] = 1; }
        }
    }
    else if ( state === BRAIN_STATE_PURSUING_PREY )
    {
        // meat it touches while pursuing prey is eaten first
        const meatX = f >= 0 ? st.foodX[f] - mouthX : 0, meatY = f >= 0 ? st.foodY[f] - mouthY : 0;
        if ( foodIsMeatAndAlive( st, f ) && ( Math.sqrt( meatX * meatX + meatY * meatY ) < SWIMBOT_MOUTH_LENGTH + FOOD_BIT_GRAB_RADIUS ))
        {
            M[ m + SS.M_TRY_EAT ] = 1;
        }
        else if ( preyValid && ( M[ m + SS.M_HUNT_CD ] === 0 ))
        {
            if ( BITE_AT_CONTACT_ENABLED )
            {
                // the mouth must be at one of the prey's body parts
                if ( mouthTouchesBody( st, prey, mouthX, mouthY )) { M[ m + SS.M_TRY_HUNT ] = 1; }
            }
            else
            {
                const p = prey * SS.BOT_STRIDE;
                const biteDistance = SWIMBOT_MOUTH_LENGTH + B[ p + SS.B_SUM_LENGTHS ] * 0.5;
                const xx = B[ p + SS.B_POS_X ] - mouthX, yy = B[ p + SS.B_POS_Y ] - mouthY;
                if ( Math.sqrt( xx * xx + yy * yy ) < biteDistance ) { M[ m + SS.M_TRY_HUNT ] = 1; }
            }
        }
    }
    else if ( state === BRAIN_STATE_LOOKING_FOR_PREY )
    {
        // a hungry predator with no prey in sight eats meat it sees
        if ( foodIsMeatAndAlive( st, f ))
        {
            const xx = st.foodX[f] - mouthX, yy = st.foodY[f] - mouthY;
            if ( Math.sqrt( xx * xx + yy * yy ) < SWIMBOT_MOUTH_LENGTH + FOOD_BIT_GRAB_RADIUS ) { M[ m + SS.M_TRY_EAT ] = 1; }
        }
    }

    //-------------------------------------------------
    // the mouth touches something edible: eat it, whatever the brain is
    // doing (fleeing, looking for a mate...), unless already full
    //-------------------------------------------------
    if ( CONTACT_EATING_ENABLED && ( M[ m + SS.M_TRY_EAT ] !== 1 ) && ( B[ b + SS.B_ENERGY ] < M[ m + SS.M_MAX_ENERGY ] ))
    {
        const touched = edibleFoodAtMouth( st, s, mouthX, mouthY );
        if ( touched >= 0 ) { M[ m + SS.M_FOOD ] = touched; M[ m + SS.M_TRY_EAT ] = 1; }
    }

    //-------------------------------------------------
    // YOUNG_CANNIBALISM_ENABLED: a hungry herbivore bites the young one of
    // its kind (fewer body parts) it is after, when its mouth reaches it
    //-------------------------------------------------
    if ( YOUNG_CANNIBALISM_ENABLED && ( M[ m + SS.M_CANNIBAL ] === 1 ) && ( M[ m + SS.M_TRY_HUNT ] !== 1 ) && ( M[ m + SS.M_TRY_EAT ] !== 1 ))
    {
        const young = M[ m + SS.M_PREY ];
        if (( young >= 0 ) && ( st.liveSwimbot[ young ] === 1 ) && ( M[ m + SS.M_HUNT_CD ] === 0 ) && mouthTouchesBody( st, young, mouthX, mouthY ))
        {
            M[ m + SS.M_TRY_HUNT ] = 1;
        }
    }

    //-------------------------------------------------
    // the direction to the goal
    //-------------------------------------------------
    if ( state === BRAIN_STATE_LOOKING_FOR_PREY )
    {
        if ( foodIsMeatAndAlive( st, f )) { aim( B, b, st.foodX[f], st.foodY[f], mouthX, mouthY ); }
        else                              { wanderFocus( st, s ); }
    }
    else if ( state === BRAIN_STATE_PURSUING_PREY )
    {
        if ( preyValid ) { const p = prey * SS.BOT_STRIDE; aim( B, b, B[ p + SS.B_POS_X ], B[ p + SS.B_POS_Y ], mouthX, mouthY ); }
    }
    else if ( state === BRAIN_STATE_FLEEING_PREDATOR )
    {
        const nearPred = M[ m + SS.M_NEAR_PRED ];
        const danger = ( nearPred >= 0 ) ? nearPred : M[ m + SS.M_CALM_PRED ];
        if (( danger >= 0 ) && ( st.liveSwimbot[ danger ] === 1 ))
        {
            // swim directly away from the predator
            const d = danger * SS.BOT_STRIDE;
            aim( B, b, B[ b + SS.B_POS_X ], B[ b + SS.B_POS_Y ], B[ d + SS.B_POS_X ], B[ d + SS.B_POS_Y ] );
        }
    }
    else if (( state === BRAIN_STATE_LOOKING_FOR_FOOD ) && ( M[ m + SS.M_CANNIBAL ] === 1 ) && ( M[ m + SS.M_PREY ] >= 0 ) && ( st.liveSwimbot[ M[ m + SS.M_PREY ] ] === 1 ))
    {
        const p = M[ m + SS.M_PREY ] * SS.BOT_STRIDE;
        aim( B, b, B[ p + SS.B_POS_X ], B[ p + SS.B_POS_Y ], mouthX, mouthY );
    }
    else if (( state === BRAIN_STATE_LOOKING_FOR_FOOD ) || ( state === BRAIN_STATE_LOOKING_FOR_MATE ))
    {
        wanderFocus( st, s );
    }
    else if ( state == BRAIN_STATE_PURSUING_MATE )
    {
        if ( mate >= 0 ) { aim( B, b, st.genitalX[ mate ], st.genitalY[ mate ], B[ b + SS.B_GENITAL_X ], B[ b + SS.B_GENITAL_Y ] ); }
    }
    else if ( state === BRAIN_STATE_SELF_FEED )
    {
        B[ b + SS.B_GOAL_X ] = B[ b + SS.B_GOAL_Y ] = 0;
    }
    else if ( state === BRAIN_STATE_PURSUING_FOOD )
    {
        if ( f >= 0 ) { aim( B, b, st.foodX[f], st.foodY[f], mouthX, mouthY ); }
    }

    //-------------------------------------------------
    // neural brain: while searching, also follow the better temperature and
    // keep close to (or away from) the others, as the brain decides
    //-------------------------------------------------
    if ( M[ m + SS.M_NEURAL ] === 1 )
    {
        if (( state === BRAIN_STATE_LOOKING_FOR_FOOD ) || ( state === BRAIN_STATE_LOOKING_FOR_MATE ) || ( state === BRAIN_STATE_LOOKING_FOR_PREY ))
        {
            B[ b + SS.B_GOAL_X ] += CLIMATE_STEERING * M[ m + SS.M_CLIMATE_W ] * M[ m + SS.M_CLIMATE_X ] + SOCIAL_STEERING * M[ m + SS.M_SOCIAL_W ] * M[ m + SS.M_SOCIAL_X ];
            B[ b + SS.B_GOAL_Y ] += CLIMATE_STEERING * M[ m + SS.M_CLIMATE_W ] * M[ m + SS.M_CLIMATE_Y ] + SOCIAL_STEERING * M[ m + SS.M_SOCIAL_W ] * M[ m + SS.M_SOCIAL_Y ];
            normalize( B, b + SS.B_GOAL_X );

            // the brain may also turn the swimbot a little, left or right
            const steer = M[ m + SS.M_STEER ];
            if ( steer !== 0 )
            {
                const c = Math.cos( steer * 0.1 ), sn = Math.sin( steer * 0.1 );   // (a tenth per step: a smooth turn)
                const gx = B[ b + SS.B_GOAL_X ], gy = B[ b + SS.B_GOAL_Y ];
                B[ b + SS.B_GOAL_X ] = gx * c - gy * sn;
                B[ b + SS.B_GOAL_Y ] = gx * sn + gy * c;
            }
        }
    }

    //-------------------------------------------------
    // continually push the focus direction towards the goal
    //-------------------------------------------------
    const previousX = B[ b + SS.B_FOCUS_X ], previousY = B[ b + SS.B_FOCUS_Y ];
    const goalX = B[ b + SS.B_GOAL_X ], goalY = B[ b + SS.B_GOAL_Y ];
    let fx = previousX + goalX * BRAIN_FOCUS_TARGET_SHIFT_STRENGTH;
    let fy = previousY + goalY * BRAIN_FOCUS_TARGET_SHIFT_STRENGTH;
    const ddx = fx - previousX, ddy = fy - previousY;
    if ( ddx * ddx + ddy * ddy > BRAIN_FOCUS_TARGET_SHIFT_THRESHOLD * BRAIN_FOCUS_TARGET_SHIFT_THRESHOLD )
    {
        fx = previousX + goalX * BRAIN_FOCUS_TARGET_SHIFT_THRESHOLD;
        fy = previousY + goalY * BRAIN_FOCUS_TARGET_SHIFT_THRESHOLD;
    }
    B[ b + SS.B_FOCUS_X ] = fx;
    B[ b + SS.B_FOCUS_Y ] = fy;
    normalize( B, b + SS.B_FOCUS_X );

    //-------------------------------------------------
    // what the predator wants from a target (for the perception)
    //-------------------------------------------------
    const o = s * NUM_OUTPUTS, out = st.nnOutputs;
    st.preyValueWeight [s] = sigmoid( out[ o + OUT_PREY_VALUE ] );
    st.preyTravelWeight[s] = sigmoid( out[ o + OUT_PREY_TRAVEL ] );
    st.preyBaitBias    [s] = Math.tanh( out[ o + OUT_PREY_BAIT ] );

    //-------------------------------------------------
    // the climate costs energy
    //-------------------------------------------------
    if ( st.params[ SS.PARAM_ENV_ACTIVE ] === 1 )
    {
        updateBodyTemperature( st, s );
        if ((( age + index ) % THERMAL_FIT_UPDATE_PERIOD ) === 0 ) { updateThermalFit( st, s ); }
        B[ b + SS.B_ENERGY ] -= THERMAL_MISMATCH_DRAIN * ( 1 - M[ m + SS.M_THERMAL_FIT ] );
    }
}

//------------------------------------------------------------------------------
// after the physics: does it look around this step? (for the perception)
//------------------------------------------------------------------------------
export function prepareLook( st: SharedState, s: number ): void
{
    const M = st.mind, m = s * SS.MIND_STRIDE, B = st.bots, b = s * SS.BOT_STRIDE;
    if ( M[ m + SS.M_READY ] === 1 )
    {
        // scan radius: a hunting predator looks farther and farther
        const state = M[ m + SS.M_STATE ];
        const hunting = isPredator( st, s ) && (( state === BRAIN_STATE_LOOKING_FOR_PREY ) || ( state === BRAIN_STATE_PURSUING_PREY ));
        if ( M[ m + SS.M_SCAN_CD ] > 0 )
        {
            M[ m + SS.M_SCAN_CD ] -= BRAIN_SENSORY_UPDATE_PERIOD;
        }
        else if ( hunting && ( state === BRAIN_STATE_LOOKING_FOR_PREY ))
        {
            M[ m + SS.M_SCAN_RADIUS ] = Math.min( M[ m + SS.M_SCAN_RADIUS ] + PREY_SCAN_RADIUS_GROWTH, SWIMBOT_VIEW_RADIUS * PREY_SCAN_RADIUS_MAX_SCALE );
        }
        else if ( !hunting )
        {
            M[ m + SS.M_SCAN_RADIUS ] = Math.max( M[ m + SS.M_SCAN_RADIUS ] - PREY_SCAN_RADIUS_GROWTH, SWIMBOT_VIEW_RADIUS );
        }
        st.sensing[s] = 1;
    }
    st.mouthX    [s] = B[ b + SS.B_MOUTH_X ];
    st.mouthY    [s] = B[ b + SS.B_MOUTH_Y ];
    st.energy    [s] = B[ b + SS.B_ENERGY  ];
    st.scanRadius[s] = M[ m + SS.M_SCAN_RADIUS ];
    st.brainState[s] = M[ m + SS.M_STATE ];
    st.age       [s] = B[ b + SS.B_AGE ];
}

//------------------------------------------------------------------------------
// after looking around: what it saw becomes its targets
// (was GenePool.applyPerception and Swimbot.setEnvironmentalStimuli)
//------------------------------------------------------------------------------
export function takeInLook( st: SharedState, s: number, counters: number ): void
{
    const M = st.mind, m = s * SS.MIND_STRIDE, B = st.bots, b = s * SS.BOT_STRIDE;
    const neural = M[ m + SS.M_NEURAL ] === 1;
    const count  = st.seenCount[s];
    // (a bite decided at the start of the step stays on that food bit)
    const eatingNow = ( M[ m + SS.M_TRY_EAT ] === 1 ) ? M[ m + SS.M_FOOD ] : -1;
    const base   = s * SS.MAX_SEEN;

    if ( neural )
    {
        const sx = st.socialX[s], sy = st.socialY[s];
        const length = Math.sqrt( sx * sx + sy * sy );
        M[ m + SS.M_SOCIAL_X ] = ( length > 0 ) ? sx / length : 0;
        M[ m + SS.M_SOCIAL_Y ] = ( length > 0 ) ? sy / length : 0;
    }
    if ( PREDATOR_TARGET_SELECTION_ENABLED )
    {
        const x = st.nnInputs, xi = s * NUM_INPUTS;
        x[ xi + IN_PREY_DISTANCE ] = st.targetPreyDistance[s] || 0;
        x[ xi + IN_PREY_REWARD   ] = st.targetPreyReward[s]   || 0;
        x[ xi + IN_PREY_SUCCESS  ] = st.targetPreySuccess[s]  || 0;
        x[ xi + IN_PREY_CHASE    ] = st.targetPreyChase[s]    || 0;
        x[ xi + IN_BAIT_PRESENT  ] = ( st.targetBaitPresent[s] === 1 ) ? 1 : 0;
        x[ xi + IN_BAIT_DISTANCE ] = st.targetBaitDistance[s] || 0;
        x[ xi + IN_BAIT_REWARD   ] = st.targetBaitReward[s]   || 0;
    }

    // (something may have died or been eaten since the look)
    const f = st.seenFood[s], p = st.seenPrey[s], t = st.seenThreat[s], a = st.seenPredator[s];
    const foodOkay   = ( f >= 0 ) && ( st.liveFood[f] === 1 );
    const preyOkay   = ( p >= 0 ) && ( st.liveSwimbot[p] === 1 );
    const threatOkay = ( t >= 0 ) && ( st.liveSwimbot[t] === 1 );
    const anyPredator = (( a >= 0 ) && ( st.liveSwimbot[a] === 1 )) ? a : -1;
    const foodIsMeat = foodOkay && ( st.foodType[f] === 2 );
    const K = st.counters, k = counters;

    if ( PREDATOR_TARGET_SELECTION_ENABLED && ( preyOkay || foodIsMeat ))
    {
        K[ k + SS.K_PREY_CHOICES ] ++;
        if ( foodIsMeat && !preyOkay )
        {
            K[ k + SS.K_PREY_MEAT_CHOICES  ] ++;
            K[ k + SS.K_PREY_MEAT_DISTANCE ] += st.targetBaitDistance[s];
            K[ k + SS.K_PREY_MEAT_REWARD   ] += st.targetBaitReward[s];
            K[ k + SS.K_PREY_DISTANCE      ] += st.targetBaitDistance[s];
            K[ k + SS.K_PREY_REWARD        ] += st.targetBaitReward[s];
        }
        else
        {
            K[ k + SS.K_PREY_LIVE_CHOICES  ] ++;
            K[ k + SS.K_PREY_LIVE_DISTANCE ] += st.targetPreyDistance[s];
            K[ k + SS.K_PREY_LIVE_REWARD   ] += st.targetPreyReward[s] * st.targetPreySuccess[s];
            K[ k + SS.K_PREY_DISTANCE      ] += st.targetPreyDistance[s];
            K[ k + SS.K_PREY_REWARD        ] += st.targetPreyReward[s] * st.targetPreySuccess[s];
        }
    }

    if ( preyOkay ) { M[ m + SS.M_SCAN_CD ] = PREY_SCAN_COOLDOWN; }

    //------------------------------------------------------------
    // remember what I see, for the neural brain
    //------------------------------------------------------------
    M[ m + SS.M_SEEN_FOOD ] = ( foodOkay && !foodIsMeat ) ? 1 : 0;
    M[ m + SS.M_SEEN_PREY ] = ( preyOkay || foodIsMeat ) ? 1 : 0;
    M[ m + SS.M_NUM_SEEN  ] = count;
    if ( foodOkay && !foodIsMeat ) { M[ m + SS.M_SEEN_FOOD_X ] = st.foodX[f]; M[ m + SS.M_SEEN_FOOD_Y ] = st.foodY[f]; }
    M[ m + SS.M_NEAREST_SEEN ] = ( count > 0 ) ? st.seen[ base ] : -1;

    //------------------------------------------------------------
    // danger: a hunting predator close enough makes me flee
    //------------------------------------------------------------
    const px = B[ b + SS.B_POS_X ], py = B[ b + SS.B_POS_Y ];
    let nearPred = -1;
    M[ m + SS.M_THREAT_CLOSE ] = 0;
    M[ m + SS.M_PRED_CLOSE ]   = 0;
    if ( threatOkay )
    {
        const d = t * SS.BOT_STRIDE, dx = B[ d + SS.B_POS_X ] - px, dy = B[ d + SS.B_POS_Y ] - py;
        const distance = Math.sqrt( dx * dx + dy * dy );
        M[ m + SS.M_THREAT_CLOSE ] = Math.max( 0, 1 - distance / SWIMBOT_VIEW_RADIUS );
        if ( distance < PREDATOR_FLEE_RADIUS ) { nearPred = t; }
    }
    M[ m + SS.M_NEAR_PRED ] = nearPred;
    M[ m + SS.M_THREAT_NEAR ] = ( nearPred >= 0 ) ? 1 : 0;

    // autotomy as a decoy: a hunter came too close - drop a part (main thread)
    const autotomyThreat = M[ m + SS.M_AUTOTOMY_THREAT ];
    if ( threatOkay && ( M[ m + SS.M_PARTS_LOST ] !== 1 ) && ( autotomyThreat > 0.05 )
    &&   ( M[ m + SS.M_THREAT_CLOSE ] >= 1 - AUTOTOMY_DECOY_SPAN * autotomyThreat ))
    {
        M[ m + SS.M_EVENTS ] |= SS.EVENT_DECOY;
    }

    if ( anyPredator >= 0 )
    {
        const d = anyPredator * SS.BOT_STRIDE, dx = B[ d + SS.B_POS_X ] - px, dy = B[ d + SS.B_POS_Y ] - py;
        M[ m + SS.M_PRED_CLOSE ] = Math.max( 0, 1 - Math.sqrt( dx * dx + dy * dy ) / SWIMBOT_VIEW_RADIUS );
        M[ m + SS.M_CALM_PRED ] = ( neural && ( nearPred < 0 )) ? anyPredator : -1;
    }
    else { M[ m + SS.M_CALM_PRED ] = -1; }

    //------------------------------------------------------------
    // if hunting, choose the prey that was found
    //------------------------------------------------------------
    const state = M[ m + SS.M_STATE ];
    const huntingState = ( state == BRAIN_STATE_LOOKING_FOR_PREY ) || ( state == BRAIN_STATE_PURSUING_PREY );
    M[ m + SS.M_PREY ] = -1;
    if ( huntingState )
    {
        M[ m + SS.M_FOUND_PREY ] = preyOkay ? 1 : 0;
        // a chosen meat target replaces the live-prey chase
        if ( PREDATOR_TARGET_SELECTION_ENABLED && !preyOkay && foodIsMeat && ( state === BRAIN_STATE_PURSUING_PREY ))
        {
            M[ m + SS.M_STATE ] = BRAIN_STATE_LOOKING_FOR_PREY;
        }
        if ( preyOkay ) { M[ m + SS.M_PREY ] = p; }
    }

    //------------------------------------------------------------
    // keep a meat target available while a predator hunts
    //------------------------------------------------------------
    const stateNow = M[ m + SS.M_STATE ];
    M[ m + SS.M_FOOD ] = -1;
    if ((( stateNow == BRAIN_STATE_LOOKING_FOR_PREY ) || ( stateNow == BRAIN_STATE_PURSUING_PREY )) && foodIsMeat )
    {
        M[ m + SS.M_FOOD ] = f;
    }
    if (( stateNow === BRAIN_STATE_PURSUING_PREY ) && ( M[ m + SS.M_FOOD ] >= 0 ))
    {
        if ( M[ m + SS.M_FOOD ] !== M[ m + SS.M_LAST_MEAT ] ) K[ k + SS.K_MEAT_TARGETS_PURSUING ] ++;
        M[ m + SS.M_LAST_MEAT ] = M[ m + SS.M_FOOD ];
    }
    else { M[ m + SS.M_LAST_MEAT ] = -1; }

    if (( stateNow == BRAIN_STATE_LOOKING_FOR_FOOD ) || ( stateNow == BRAIN_STATE_PURSUING_FOOD ))
    {
        M[ m + SS.M_FOUND_FOOD ] = foodOkay ? 1 : 0;
        if ( foodOkay ) { M[ m + SS.M_FOOD ] = f; }
    }

    // nothing to eat anymore - cancel any pending bite
    if ( M[ m + SS.M_FOOD ] < 0 ) { M[ m + SS.M_TRY_EAT ] = 0; }

    //------------------------------------------------------------
    // YOUNG_CANNIBALISM_ENABLED: a hungry herbivore that sees no plant may go
    // for a young one of its kind with fewer body parts (the nearest)
    //------------------------------------------------------------
    M[ m + SS.M_CANNIBAL ] = 0;
    if ( YOUNG_CANNIBALISM_ENABLED && !isPredator( st, s ) && !( foodOkay && !foodIsMeat )
    &&   ( B[ b + SS.B_ENERGY ] < M[ m + SS.M_HUNGER ] ))
    {
        const myParts = B[ b + SS.B_NUM_PARTS ];
        for (let i = 0; i < count; i++)
        {
            const o = st.seen[ base + i ];
            if (( st.liveSwimbot[o] !== 1 ) || isPredator( st, o ) || ( st.eternal[o] === 1 )) continue;
            if ( st.age[o] >= st.mind[ o * SS.MIND_STRIDE + SS.M_MATURE_AGE ] ) continue;           // only the young
            if ( st.bots[ o * SS.BOT_STRIDE + SS.B_NUM_PARTS ] >= myParts ) continue;              // with fewer parts
            M[ m + SS.M_PREY ] = o;
            M[ m + SS.M_CANNIBAL ] = 1;
            break;
        }
    }

    //------------------------------------------------------------
    // a mate
    //------------------------------------------------------------
    if ( stateNow === BRAIN_STATE_LOOKING_FOR_MATE )
    {
        const chosen = st.seenMate[s];
        if ( chosen === -2 )
        {
            // other criteria than color: the main thread chooses (EVENT_CHOOSE_MATE)
            M[ m + SS.M_EVENTS ] |= SS.EVENT_CHOOSE_MATE;
        }
        else if (( chosen >= 0 ) && ( st.liveSwimbot[ chosen ] === 1 ))
        {
            M[ m + SS.M_MATE ] = chosen;
            M[ m + SS.M_FOUND_MATE ] = 1;
        }
        else
        {
            M[ m + SS.M_FOUND_MATE ] = 0;
        }
    }
    else if ( stateNow == BRAIN_STATE_PURSUING_MATE )
    {
        // can I still see my mate? (a mate that changed its diet is given up)
        const mate = M[ m + SS.M_MATE ];
        let stillSeen = false;
        for (let i = 0; i < count; i++)
        {
            const o = st.seen[ base + i ];
            if (( o === mate ) && ( isPredator( st, o ) === isPredator( st, s ))) { stillSeen = true; }
        }
        if ( !stillSeen )
        {
            M[ m + SS.M_FOUND_MATE ] = 0;
            M[ m + SS.M_MATE ] = -1;
        }
    }

    if (( eatingNow >= 0 ) && ( st.liveFood[ eatingNow ] === 1 ))
    {
        M[ m + SS.M_FOOD ] = eatingNow;
        M[ m + SS.M_TRY_EAT ] = 1;
    }

    M[ m + SS.M_READY ] = 0;
}

//------------------------------------------------------------------------------
// Perception kernel (stage 7): what one swimbot sees - the nearest swimbots,
// the best food bit, prey (for predators), the nearest hunting predator (for
// herbivores). It only reads the shared arrays (positions as they are now)
// and writes its answers into the swimbot's output slots, so many swimbots
// can look around at the same time in different threads.
//
// The arithmetic is the same as the old GenePool.giveSwimbotNearbyEnvironmentalStimuli
// (checked by comparing both on the same state, see GenePool PERCEPTION_CHECK).
//------------------------------------------------------------------------------
import { BRAIN_STATE_LOOKING_FOR_MATE, BRAIN_STATE_LOOKING_FOR_PREY, BRAIN_STATE_PURSUING_PREY } from "./Brain";
import { IGNORE_FOOD_EFFICIENCY, mateAcceptance, pairRandom, preyVisibility, searchImageFactor, sizeSignature } from "./Ecology";
import { PLANT_TYPE_WARM } from "./Environment";
import { ONE, ONE_THIRD } from "./MathConstants";
import { ATTACK_SUCCESS_CHANCE, ENVIRONMENT_ATTACK_SUCCESS_CHANCE, PREDATOR_LEVEL_THRESHOLD, PREY_MAX_MASS_RATIO, PREY_SCAN_COOLDOWN, SWIMBOT_MAX_ENERGY, PREDATION_ENERGY_TRANSFER, PREY_BODY_ENERGY, YOUNG_AGE_DURATION, PREDATOR_TARGET_SELECTION_ENABLED } from "./Parameters";
import { POOL_BOTTOM, POOL_LEFT, POOL_RIGHT, POOL_TOP } from "./Pool";
import { ATTRACTION_SIMILAR_COLOR, STARVING, SWIMBOT_MOUTH_LENGTH, SWIMBOT_VIEW_RADIUS } from "./SwimbotTypes";
import { FOOD_BIT_GRAB_RADIUS } from "./FoodBit";
import { JUVENILE_PREDATION_ENABLED, CROWD_AWARE_FOOD_CHOICE_ENABLED, FOOD_TARGET_PERSISTENCE_ENABLED, FOOD_SWITCH_RATIO_MIN, FOOD_KEEP_RADIUS_SCALE, SIZE_ASSORTATIVE_MATING_ENABLED } from "./Parameters";
import { MIND_STRIDE, M_DEFERENCE, M_FOOD, M_PERSIST, M_MATURE_AGE, M_SIZE_PICKY, M_MAX_ENERGY } from "./MindLayout";
import { preyTargetScore } from "./EcologyGenes";
import {
    BACKGROUND_SIZE, B_VEL_X, B_VEL_Y, BOT_STRIDE, MAX_SEEN, PARAM_ENV_ACTIVE, PARAM_ENV_FIELDS, PARAM_FOOD_TYPES, PARAM_OBSTACLE_X1, PARAM_OBSTACLE_X2,
    PARAM_OBSTACLE_Y1, PARAM_OBSTACLE_Y2, SharedState
} from "./SharedState";

export const STARVING_FOOD_CHOICE = 15;   // below this energy a swimbot eats anything

//------------------------------------------------------------
// scratch arrays: one set per thread
//------------------------------------------------------------
export class PerceptionScratch
{
    readonly gridResult         : Int32Array;
    readonly foodGridResult     : Int32Array;
    readonly nearbyIndices      : Int32Array;
    readonly candidateIndices   : Int32Array;
    readonly candidateDistances : Float64Array;
    readonly nearestIndices     = new Int32Array  ( MAX_SEEN );
    readonly nearestDistances   = new Float64Array( MAX_SEEN );

    constructor( maxSwimbots: number, maxFoodBits: number )
    {
        this.gridResult         = new Int32Array  ( maxSwimbots * 2 );
        this.foodGridResult     = new Int32Array  ( maxFoodBits * 2 );
        this.nearbyIndices      = new Int32Array  ( maxSwimbots );
        this.candidateIndices   = new Int32Array  ( maxSwimbots );
        this.candidateDistances = new Float64Array( maxSwimbots );
    }
}

//------------------------------------------------------------
// helpers (same as in GenePool / Vector2D)
//------------------------------------------------------------
function isCloser( distanceA: number, indexA: number, distanceB: number, indexB: number ): boolean
{
    return ( distanceA < distanceB ) || ( distanceA === distanceB && indexA < indexB );
}

function sortCandidatesByDistance( count: number, sc: PerceptionScratch ): void
{
    const ci = sc.candidateIndices, cd = sc.candidateDistances;
    for (let i = 1; i < count; i++)
    {
        const index = ci[i];
        const d     = cd[i];
        let j = i - 1;
        while ( j >= 0 && isCloser( d, index, cd[j], ci[j] ))
        {
            ci[j + 1] = ci[j];
            cd[j + 1] = cd[j];
            j--;
        }
        ci[j + 1] = index;
        cd[j + 1] = d;
    }
}

function selectNearestCandidates( count: number, limit: number, sc: PerceptionScratch ): number
{
    const ci = sc.candidateIndices, cd = sc.candidateDistances;
    const ni = sc.nearestIndices,   nd = sc.nearestDistances;
    let kept = 0;
    for (let i = 0; i < count; i++)
    {
        const index = ci[i];
        const d     = cd[i];

        if (( kept === limit ) && !isCloser( d, index, nd[ kept - 1 ], ni[ kept - 1 ] )) continue;

        let j = ( kept < limit ) ? kept : kept - 1;
        if ( kept < limit ) kept ++;

        while ( j > 0 && isCloser( d, index, nd[j - 1], ni[j - 1] ))
        {
            ni[j] = ni[j - 1];
            nd[j] = nd[j - 1];
            j--;
        }
        ni[j] = index;
        nd[j] = d;
    }
    return kept;
}

// is the line from (a0x, a0y) to (a1x, a1y) blocked by the obstacle?
function obstructed( st: SharedState, a0x: number, a0y: number, a1x: number, a1y: number ): boolean
{
    const P = st.params;
    const b0x = P[ PARAM_OBSTACLE_X1 ], b0y = P[ PARAM_OBSTACLE_Y1 ];
    const b1x = P[ PARAM_OBSTACLE_X2 ], b1y = P[ PARAM_OBSTACLE_Y2 ];

    const aX = a1x - a0x, aY = a1y - a0y;
    const bX = b1x - b0x, bY = b1y - b0y;
    const aPerpX = -aY, aPerpY = aX;
    const bPerpX = -bY, bPerpY = bX;
    const a0b0x = b0x - a0x, a0b0y = b0y - a0y;
    const a0b1x = b1x - a0x, a0b1y = b1y - a0y;
    const b0a0x = a0x - b0x, b0a0y = a0y - b0y;
    const b0a1x = a1x - b0x, b0a1y = a1y - b0y;
    const a0Dotb0 = aPerpX * a0b0x + aPerpY * a0b0y;
    const a0Dotb1 = aPerpX * a0b1x + aPerpY * a0b1y;
    const b0Dota0 = bPerpX * b0a0x + bPerpY * b0a0y;
    const b0Dota1 = bPerpX * b0a1x + bPerpY * b0a1y;

    return (((( a0Dotb0 > 0 ) && ( a0Dotb1 < 0 )) || (( a0Dotb1 > 0 ) && ( a0Dotb0 < 0 )))
        &&  ((( b0Dota0 > 0 ) && ( b0Dota1 < 0 )) || (( b0Dota1 > 0 ) && ( b0Dota0 < 0 ))));
}

// brightness of the bottom (as Environment.backgroundAt)
function backgroundAt( st: SharedState, x: number, y: number ): number
{
    if ( st.params[ PARAM_ENV_FIELDS ] === 0 ) return 0.5;
    const n = BACKGROUND_SIZE;
    const i = Math.min( n - 1, Math.max( 0, Math.floor( ( x - POOL_LEFT ) / ( POOL_RIGHT  - POOL_LEFT ) * n )));
    const j = Math.min( n - 1, Math.max( 0, Math.floor( ( y - POOL_TOP  ) / ( POOL_BOTTOM - POOL_TOP  ) * n )));
    return st.background[ j * n + i ];
}

//------------------------------------------------------------------------------
// what swimbot s sees now
//------------------------------------------------------------------------------
export function perceive( s: number, st: SharedState, sc: PerceptionScratch ): void
{
    const viewRadius            = SWIMBOT_VIEW_RADIUS;
    const viewRadiusSquared     = viewRadius * viewRadius;
    const preyViewRadius        = st.scanRadius[s];
    const predatorLevel         = st.predator[s];
    const myMass                = st.mass[s];
    const myX                   = st.botX[s];
    const myY                   = st.botY[s];
    const myGenitalX            = st.genitalX[s];
    const myGenitalY            = st.genitalY[s];
    const myMouthX              = st.mouthX[s];
    const myMouthY              = st.mouthY[s];
    const iAmPredator           = predatorLevel > PREDATOR_LEVEL_THRESHOLD;

    const botX = st.botX, botY = st.botY, genitalX = st.genitalX, genitalY = st.genitalY;
    const alive = st.alive, predator = st.predator;

    //------------------------------------------------------------------
    // everybody within the search radius (wider for a hunting predator)
    //------------------------------------------------------------------
    const searchRadius        = iAmPredator ? preyViewRadius : viewRadius;
    const searchRadiusSquared = searchRadius * searchRadius;
    const nearby = sc.nearbyIndices;
    let numNearby = 0;
    const numInCells = st.swimbotGrid.collect( myX, myY, searchRadius, sc.gridResult );

    for (let k = 0; k < numInCells; k++)
    {
        const o = sc.gridResult[k];
        if ( o === s ) continue;
        const dx = botX[o] - myX;
        const dy = botY[o] - myY;
        if ( dx * dx + dy * dy < searchRadiusSquared ) { nearby[ numNearby ++ ] = o; }
    }

    //------------------------------------------------------------------
    // the NEAREST swimbots within normal view, in order of distance
    //------------------------------------------------------------------
    let numCandidates = 0;
    for (let i = 0; i < numNearby; i++)
    {
        const o = nearby[i];
        if ( alive[o] === 1 )
        {
            const dx = myGenitalX - genitalX[o];
            const dy = myGenitalY - genitalY[o];
            const distanceSquared = dx * dx + dy * dy;
            if ( distanceSquared < viewRadiusSquared )
            {
                sc.candidateIndices  [ numCandidates ] = o;
                sc.candidateDistances[ numCandidates ] = distanceSquared;
                numCandidates ++;
            }
        }
    }

    let numNearest = selectNearestCandidates( numCandidates, MAX_SEEN, sc );
    let nearestList: Int32Array = sc.nearestIndices;

    for (let i = 0; i < numNearest; i++)
    {
        const o = sc.nearestIndices[i];
        if ( obstructed( st, myGenitalX, myGenitalY, genitalX[o], genitalY[o] ) && ( numCandidates > numNearest ))
        {
            sortCandidatesByDistance( numCandidates, sc );
            nearestList = sc.candidateIndices;
            numNearest  = numCandidates;
            break;
        }
    }

    const seenBase = s * MAX_SEEN;
    let numSeen = 0;
    let socialX = 0, socialY = 0;
    for (let i = 0; i < numNearest && numSeen < MAX_SEEN; i++)
    {
        const o = nearestList[i];
        if ( !obstructed( st, myGenitalX, myGenitalY, genitalX[o], genitalY[o] ))
        {
            st.seen[ seenBase + numSeen ] = o;
            numSeen ++;
            socialX += botX[o] - myX;
            socialY += botY[o] - myY;
        }
    }
    st.seenCount[s] = numSeen;
    st.socialX[s]   = socialX;
    st.socialY[s]   = socialY;

    //------------------------------------------------------------------
    // the best food bit
    //------------------------------------------------------------------
    const environmentOn    = st.params[ PARAM_ENV_ACTIVE ] !== 0;
    const myEfficiencyWarm = environmentOn ? st.effWarm[s] : ONE;
    const myEfficiencyCold = environmentOn ? st.effCold[s] : ONE;
    const iAmStarving      = st.energy[s] < STARVING_FOOD_CHOICE;
    const twoFoodTypes     = st.params[ PARAM_FOOD_TYPES ] === 2;
    const preferredFoodType = st.preferred[s];
    const foodX = st.foodX, foodY = st.foodY, foodType = st.foodType, foodAlive = st.foodAlive;
    const botOffset = s * BOT_STRIDE;
    const swimSpeed = Math.max( 0.05, Math.hypot( st.bots[ botOffset + B_VEL_X ], st.bots[ botOffset + B_VEL_Y ] ));

    let chosenFood = -1;
    let smallestDistanceSquared = Number.MAX_SAFE_INTEGER;
    let bestMeatScore = -Infinity;
    let baitFound = false;
    let baitDistance = 0, baitReward = 0;
    // Search far enough from the body center to include food touching the
    // forward mouth, then apply the normal view-radius test from the mouth.
    const foodSearchRadius = viewRadius + SWIMBOT_MOUTH_LENGTH + FOOD_BIT_GRAB_RADIUS;
    const numFoodInCells = st.foodGrid.collect( myX, myY, foodSearchRadius, sc.foodGridResult );

    for (let k = 0; k < numFoodInCells; k++)
    {
        const f  = sc.foodGridResult[k];
        const fx = foodX[f] - myX;
        const fy = foodY[f] - myY;
        const type = foodType[f];
        const bodyDistanceSquared = fx * fx + fy * fy;
        const mx = myMouthX - foodX[f];
        const my = myMouthY - foodY[f];
        const viewDistanceSquared = mx * mx + my * my;
        if ( type === 2 )
        {
            if ( viewDistanceSquared >= viewRadiusSquared ) continue;
        }
        else if ( bodyDistanceSquared >= viewRadiusSquared ) continue;

        let okay = true;

        if ( type === 2 )
        {
            if ( !iAmPredator ) okay = false;            // meat: predators only
        }
        else if ( iAmPredator )
        {
            okay = false;                                // predators don't eat plants
        }
        else if ( environmentOn )
        {
            if ((( type === PLANT_TYPE_WARM ) ? myEfficiencyWarm : myEfficiencyCold ) < IGNORE_FOOD_EFFICIENCY && !iAmStarving ) okay = false;
        }
        else if ( twoFoodTypes )
        {
            if ( type != preferredFoodType ) okay = false;
        }
        if ( !okay ) continue;

        if ( type === 2 && foodAlive[f] === 1 && PREDATOR_TARGET_SELECTION_ENABLED )
        {
            const room = Math.max( 0, st.mind[ s * MIND_STRIDE + M_MAX_ENERGY ] - st.energy[s] );
            const reward = Math.min( room, st.foodEnergy[f] ) / SWIMBOT_MAX_ENERGY;
            const distanceCost = Math.sqrt( viewDistanceSquared ) / Math.max( 1, viewRadius );
            const chaseCost = Math.min( 1, Math.sqrt( viewDistanceSquared ) / swimSpeed / Math.max( 1, PREY_SCAN_COOLDOWN ));
            const score = preyTargetScore( reward, chaseCost, st.preyValueWeight[s],
                                           st.preyTravelWeight[s], st.preyBaitBias[s], true );
            if ( score > bestMeatScore && !obstructed( st, myMouthX, myMouthY, foodX[f], foodY[f] ))
            {
                bestMeatScore = score;
                baitFound = true;
                chosenFood = f;
                baitDistance = distanceCost;
                baitReward = reward;
            }
            continue;
        }

        // in the environment, better food counts as nearer
        if ( environmentOn && ( type !== 2 ))
        {
            if ( viewDistanceSquared >= viewRadiusSquared ) continue;
            const e = ( type === PLANT_TYPE_WARM ) ? myEfficiencyWarm : myEfficiencyCold;
            let scaled = viewDistanceSquared / ( e * e );
            if ( CROWD_AWARE_FOOD_CHOICE_ENABLED )
            {
                // herbivores in sight whose mouth is nearer to this plant than mine
                let rivals = 0;
                for (let i = 0; i < numSeen; i++)
                {
                    const o = st.seen[ seenBase + i ];
                    if ( predator[o] > PREDATOR_LEVEL_THRESHOLD ) continue;
                    const ox = st.mouthX[o] - foodX[f], oy = st.mouthY[o] - foodY[f];
                    if ( ox * ox + oy * oy < viewDistanceSquared ) rivals ++;
                }
                const k = 1 + st.mind[ s * MIND_STRIDE + M_DEFERENCE ] * rivals;
                scaled *= k * k;
            }
            if ( scaled < smallestDistanceSquared && foodAlive[f] === 1 && !obstructed( st, myMouthX, myMouthY, foodX[f], foodY[f] ))
            {
                smallestDistanceSquared = scaled;
                chosenFood = f;
            }
            continue;
        }

        if ( viewDistanceSquared < viewRadiusSquared && viewDistanceSquared < smallestDistanceSquared && foodAlive[f] === 1 )
        {
            if ( !obstructed( st, myMouthX, myMouthY, foodX[f], foodY[f] ))
            {
                smallestDistanceSquared = viewDistanceSquared;
                chosenFood = f;
            }
        }
    }
    //------------------------------------------------------------------
    // keep the plant I am already after, unless another one is clearly
    // nearer (how clearly: the gene "persistence"); otherwise the target
    // jumped every look and most chases were never finished
    //------------------------------------------------------------------
    if ( FOOD_TARGET_PERSISTENCE_ENABLED && !iAmPredator )
    {
        const current = st.mind[ s * MIND_STRIDE + M_FOOD ];
        if (( current >= 0 ) && ( current !== chosenFood ) && ( foodAlive[ current ] === 1 ) && ( foodType[ current ] !== 2 )
        &&  !obstructed( st, myMouthX, myMouthY, foodX[ current ], foodY[ current ] ))
        {
            const cx = myMouthX - foodX[ current ], cy = myMouthY - foodY[ current ];
            const currentSquared = cx * cx + cy * cy;
            if ( currentSquared < FOOD_KEEP_RADIUS_SCALE * FOOD_KEEP_RADIUS_SCALE * viewRadiusSquared )   // (a plant once chosen is kept a little beyond sight)
            {
                let newSquared = Infinity;
                if ( chosenFood >= 0 ) { const nx = myMouthX - foodX[ chosenFood ], ny = myMouthY - foodY[ chosenFood ]; newSquared = nx * nx + ny * ny; }
                const ratio = 1 - ( 1 - FOOD_SWITCH_RATIO_MIN ) * st.mind[ s * MIND_STRIDE + M_PERSIST ];
                if ( newSquared >= ratio * ratio * currentSquared ) { chosenFood = current; }
            }
        }
    }

    st.seenFood[s] = chosenFood;
    st.targetBaitPresent[s] = baitFound ? 1 : 0;
    st.targetBaitDistance[s] = baitDistance;
    st.targetBaitReward[s] = baitReward;

    //------------------------------------------------------------------
    // prey (predators only): the nearest herbivore it can notice
    //------------------------------------------------------------------
    let chosenPrey = -1;
    if ( iAmPredator )
    {
        const imageStrength = st.imageStrength[s];
        const imageR = st.imageR[s], imageG = st.imageG[s], imageB = st.imageB[s], imageSize = st.imageSize[s];
        let bestPreyValue = -Infinity;
        let bestPreyReward = 0, bestPreyDistance = 0, bestPreySuccess = 0, bestPreyChase = 0;
        const attackChance = environmentOn ? ENVIRONMENT_ATTACK_SUCCESS_CHANCE : ATTACK_SUCCESS_CHANCE;
        for (let i = 0; i < numNearby; i++)
        {
            const o = nearby[i];
            if ( alive[o] === 0 ) continue;
            if ( predator[o] > PREDATOR_LEVEL_THRESHOLD ) continue;      // predators don't hunt predators
            if ( st.eternal[o] === 1 ) continue;                        // immortals are not eaten
            // too big to bite (a not yet grown-up one can be bitten anyway)
            const young = JUVENILE_PREDATION_ENABLED && ( st.age[o] < st.mind[ o * MIND_STRIDE + M_MATURE_AGE ] );
            if (( st.mass[o] > myMass * PREY_MAX_MASS_RATIO ) && !young ) continue;

            const px = myMouthX - botX[o];
            const py = myMouthY - botY[o];
            const distance = Math.sqrt( px * px + py * py );
            if ( distance > preyViewRadius ) continue;
            if ( obstructed( st, myMouthX, myMouthY, botX[o], botY[o] )) continue;   // (no seeing through the wall)

            // camouflage: prey that matches the bottom is seen only up close;
            // the search image: prey like the last meals is seen from farther
            if ( environmentOn )
            {
                const image = searchImageFactor( imageStrength,
                    st.colorR[o] - imageR, st.colorG[o] - imageG, st.colorB[o] - imageB, sizeSignature( st.mass[o] ) - imageSize );
                if ( distance > preyViewRadius * image * preyVisibility( st.brightness[o], backgroundAt( st, botX[o], botY[o] ))) continue;
            }

            let value = -distance + st.energy[o] * 0.001;
            if ( PREDATOR_TARGET_SELECTION_ENABLED )
            {
                const room = Math.max( 0, SWIMBOT_MAX_ENERGY - st.energy[s] );
                const potential = st.energy[o] * PREDATION_ENERGY_TRANSFER + st.mass[o] * PREY_BODY_ENERGY;
                const reward = Math.min( room, potential ) / SWIMBOT_MAX_ENERGY;
                const distanceCost = distance / Math.max( 1, preyViewRadius );
                const chaseCost = Math.min( 1, distance / swimSpeed / Math.max( 1, PREY_SCAN_COOLDOWN ));
                value = preyTargetScore( reward * attackChance, chaseCost, st.preyValueWeight[s],
                                         st.preyTravelWeight[s], st.preyBaitBias[s], false );
                if ( value > bestPreyValue )
                {
                    bestPreyReward = reward;
                    bestPreyDistance = distanceCost;
                    bestPreySuccess = attackChance;
                    bestPreyChase = chaseCost;
                }
            }
            if ( value > bestPreyValue )
            {
                bestPreyValue = value;
                chosenPrey = o;
            }
        }
        if ( PREDATOR_TARGET_SELECTION_ENABLED )
        {
            st.targetPreyReward[s] = bestPreyReward;
            st.targetPreyDistance[s] = bestPreyDistance;
            st.targetPreySuccess[s] = bestPreySuccess;
            st.targetPreyChase[s] = bestPreyChase;
            if ( bestMeatScore > bestPreyValue ) chosenPrey = -1;
            else if ( chosenPrey >= 0 && chosenFood >= 0 && foodType[ chosenFood ] === 2 )
            {
                chosenFood = -1;
                st.seenFood[s] = -1;
            }
        }
    }
    else
    {
        st.targetPreyReward[s] = st.targetPreyDistance[s] = st.targetPreySuccess[s] = st.targetPreyChase[s] = 0;
    }
    st.seenPrey[s] = chosenPrey;

    //------------------------------------------------------------------
    // the nearest hunting predator (herbivores only)
    //------------------------------------------------------------------
    // (all predators within view count, not only the nearest few swimbots:
    //  in a crowd a predator was easily "not seen")
    let threat = -1, anyPredator = -1;
    if ( !iAmPredator )
    {
        let threatDistance = Infinity, anyDistance = Infinity;
        for (let i = 0; i < numNearby; i++)
        {
            const o = nearby[i];
            if ( alive[o] === 0 ) continue;
            if ( predator[o] <= PREDATOR_LEVEL_THRESHOLD ) continue;

            const gx = myGenitalX - genitalX[o];
            const gy = myGenitalY - genitalY[o];
            const distance = Math.sqrt( gx * gx + gy * gy );
            if ( distance > viewRadius ) continue;
            if ( obstructed( st, myGenitalX, myGenitalY, genitalX[o], genitalY[o] )) continue;   // (no seeing through the wall)

            if ( distance < anyDistance ) { anyDistance = distance; anyPredator = o; }

            // the danger is a hunting predator (a fed one is calmer)
            const state = st.brainState[o];
            if (( state !== BRAIN_STATE_LOOKING_FOR_PREY ) && ( state !== BRAIN_STATE_PURSUING_PREY )) continue;
            if ( distance < threatDistance ) { threatDistance = distance; threat = o; }
        }
    }
    st.seenThreat[s]   = threat;
    st.seenPredator[s] = anyPredator;

    //------------------------------------------------------------------
    // a mate (when looking for one, by similar color): the most similar
    // grown-up, not starving swimbot in sight that accepts me
    // (as Swimbot.setEnvironmentalStimuli; other criteria: main thread)
    //------------------------------------------------------------------
    let mate = -2;
    if (( st.brainState[s] === BRAIN_STATE_LOOKING_FOR_MATE ) && ( st.attraction[s] === ATTRACTION_SIMILAR_COLOR ))
    {
        mate = -1;
        let highest = -100.0;
        const r = st.colorR, g = st.colorG, b = st.colorB;
        for (let i = 0; i < numSeen; i++)
        {
            const o = st.seen[ seenBase + i ];
            const rDiff = Math.abs( r[o] - r[s] );
            const gDiff = Math.abs( g[o] - g[s] );
            const bDiff = Math.abs( b[o] - b[s] );
            let attractiveness = ONE - ( ( rDiff + gDiff + bDiff ) * ONE_THIRD );
            if ( SIZE_ASSORTATIVE_MATING_ENABLED )
            {
                // how much a similar size counts: my own gene
                const w = st.mind[ s * MIND_STRIDE + M_SIZE_PICKY ];
                const sizeSimilarity = ONE - Math.min( ONE, Math.abs( Math.log( Math.max( 1, st.mass[o] ) / Math.max( 1, st.mass[s] ))) / 1.5 );
                attractiveness = ( ONE - w ) * attractiveness + w * sizeSimilarity;
            }

            if (( attractiveness > highest )
            &&  ( attractiveness > 0 )
            &&  ( st.age[o] > st.mind[ o * MIND_STRIDE + M_MATURE_AGE ] )
            &&  ( st.energy[o] > STARVING )
            &&  (( predator[o] > PREDATOR_LEVEL_THRESHOLD ) === iAmPredator )     // predators mate with predators only
            &&  accepts( st, o, s, environmentOn ))
            {
                highest = attractiveness;
                mate = o;
            }
        }
    }
    st.seenMate[s] = mate;
}

// does swimbot o accept swimbot s as a mate? (as Swimbot.acceptsMate)
function accepts( st: SharedState, o: number, s: number, environmentOn: boolean ): boolean
{
    if ( !environmentOn ) return true;
    const db = Math.abs( st.brightness[s] - st.brightness[o] );
    return pairRandom( s, o ) < mateAcceptance( db, st.choosiness[o] );
}

// How the environment acts on a swimbot (stage 5). Numbers: docs/БИОЛОГ_ЭТАП5.md
//
//  thermal fit       g = exp( -(T - θ)² / (2w²) )        how well the place suits me
//  thermal peak      (W_MIN / w)^a                       broad tolerance = lower peak
//  food energy       50 · 2.5 · eff(d, type) · (0.3 + 0.7 · g · peak)
//  extra drain       0.006 · (1 - g)   per step          living in the wrong climate costs
//  digestion         eff_warm = 0.15 + 0.85 · d^1.5,  eff_cold = 0.15 + 0.85 · (1-d)^1.5
//
// No genotype is best everywhere: a specialist does very well at home and badly
// elsewhere, a generalist does so-so everywhere.
//------------------------------------------------------------------------------
import { THERMAL_WIDTH_MIN } from "./EcologyGenes";
import { PLANT_TYPE_WARM } from "./Environment";

export const THERMAL_TRADE_OFF          = 0.5;     // "a": 0 = no cost of being a generalist
export const THERMAL_MISMATCH_DRAIN     = 0.006;   // extra energy per step at a totally wrong temperature
export const FOOD_BASE_FRACTION         = 0.3;     // part of the food energy you always get
// Swimbots find food at the same rate as in the classic pool, but get only a
// part of its energy here (they mostly eat "their" plant type). This bonus was
// tuned by runs (tools/eco.cjs, 3 seeds): at 1.6 the population starved down
// to ~600, at 2.2 it held 1200-2000, at 2.8 about 2600. An adapted specialist
// at home now gets ~2x the classic energy, a misplaced generalist ~0.5x.
//
// One food for everybody (TWO_PLANT_TYPES = false): digestion no longer
// matters (runs, 2 seeds x 30 000 steps: 1.75 and 2.0 slowly starved the
// population to ~1000-1500, 2.4 holds ~2000, 2.8 ~3000).
export const TWO_PLANT_TYPES            = false;
export const ENVIRONMENT_FOOD_BONUS     = TWO_PLANT_TYPES ? 2.5 : 2.4;
export const DIGESTION_MINIMUM          = 0.15;
export const DIGESTION_CURVE            = 1.5;     // > 1: in-between digestion is bad at both foods
export const IGNORE_FOOD_EFFICIENCY     = 0.3;     // food worse than this is ignored unless starving

// Body size and diet: big bodies handle the (tough) cold plants better, small
// bodies the warm ones. This ties the body form to the biome, so that one
// "best body" can't spread everywhere. size = ln(mass) relative to a typical
// swimbot (ln 370 ≈ 5.9), in units of 1.5, limited to -1..1.
export const SIZE_DIET_EFFECT           = 0.35;
const TYPICAL_LOG_MASS                  = 5.9;
const LOG_MASS_SCALE                    = 1.5;

export function bodySize( mass: number ): number
{
    const s = ( Math.log( Math.max( 1, mass )) - TYPICAL_LOG_MASS ) / LOG_MASS_SCALE;
    return s < -1 ? -1 : ( s > 1 ? 1 : s );
}

// how well a body of this size handles a plant type (meat: always 1)
export function sizeDietFactor( size: number, type: number ): number
{
    if (( type === 2 ) || !TWO_PLANT_TYPES ) return 1;
    return ( type === PLANT_TYPE_WARM ) ? 1 - SIZE_DIET_EFFECT * size : 1 + SIZE_DIET_EFFECT * size;
}

// body brightness b (step 2)
export const THERMOMELANISM             = 0.06;    // dark bodies feel warmer: T_eff = T + 0.06 · (1 - 2b)
export const PREY_DETECTION_MINIMUM     = 0.35;    // predator sees prey at R · (0.35 + 0.65 · |b - background|)
export const MATE_TOLERANCE_MINIMUM     = 0.03;    // mate acceptance width σ = 0.03 + 0.4 · (1 - choosiness)²
export const MATE_TOLERANCE_RANGE       = 0.4;

export const THERMAL_FIT_UPDATE_PERIOD  = 10;      // steps between temperature checks
export const KINESIS_PERIOD             = 50;      // steps between "is it better here?" checks
export const KINESIS_TURN_CHANCE        = 0.6;
export const KINESIS_MAX_TURN           = 120;     // degrees
export const EMERGENCY_THERMAL_FIT      = 0.2;     // below this: look around and head for the best way
export const EMERGENCY_PROBE_DISTANCE   = 300;

export function thermalFit( temperature: number, optimum: number, width: number ): number
{
    const d = temperature - optimum;
    return Math.exp( -( d * d ) / ( 2 * width * width ));
}

export function thermalPeak( width: number ): number
{
    return Math.pow( THERMAL_WIDTH_MIN / width, THERMAL_TRADE_OFF );
}

// energy multiplier of food at a place: fit = thermalFit, peak = thermalPeak
export function climateFoodFactor( fit: number, peak: number ): number
{
    return ENVIRONMENT_FOOD_BONUS * ( FOOD_BASE_FRACTION + ( 1 - FOOD_BASE_FRACTION ) * fit * peak );
}

//------------------------------------------------------------------------------
// Search image: a predator remembers what its recent prey looked like (color
// and size) and notices similar prey from farther away, different prey only
// up close. Common kinds of prey get eaten more, rare kinds less: this keeps
// many body forms and colors alive ("apostatic" selection).
//------------------------------------------------------------------------------
export const SEARCH_IMAGE_EFFECT        = 0.6;     // 0 = off; prey like the image seen up to 1 + 0.6 farther
export const SEARCH_IMAGE_LEARNING      = 0.3;     // how much one meal changes the image
export const SEARCH_IMAGE_GAIN          = 0.25;    // how much one meal strengthens it
export const SEARCH_IMAGE_FORGETTING    = 0.9997;  // per step (half gone after ~2300 steps)
export const SEARCH_IMAGE_WIDTH         = 0.18;    // how alike counts as "like the image"

// body size as a number 0..1 for the search image (ln mass around 5.9 +- 1.5)
export function sizeSignature( mass: number ): number
{
    const s = 0.5 + ( Math.log( Math.max( 1, mass )) - 5.9 ) / 3;
    return s < 0 ? 0 : ( s > 1 ? 1 : s );
}

// how much a search image changes the distance at which prey is noticed
export function searchImageFactor( strength: number, dr: number, dg: number, db: number, dsize: number ): number
{
    if ( strength <= 0 ) return 1;
    const d2 = ( dr * dr + dg * dg + db * db + dsize * dsize ) / 4;
    const similarity = Math.exp( -d2 / ( 2 * SEARCH_IMAGE_WIDTH * SEARCH_IMAGE_WIDTH ));    // 1 = just like the image
    return 1 + SEARCH_IMAGE_EFFECT * strength * ( 2 * similarity - 1 );
}

// how far away a predator notices prey (as a share of its search radius)
export function preyVisibility( brightness: number, background: number ): number
{
    return PREY_DETECTION_MINIMUM + ( 1 - PREY_DETECTION_MINIMUM ) * Math.abs( brightness - background );
}

// chance that a swimbot with this choosiness accepts a mate whose brightness differs by db
export function mateAcceptance( db: number, choosiness: number ): number
{
    const width = MATE_TOLERANCE_MINIMUM + MATE_TOLERANCE_RANGE * ( 1 - choosiness ) * ( 1 - choosiness );
    return Math.exp( -( db * db ) / ( 2 * width * width ));
}

// a fixed "random" number 0..1 for a pair of swimbots (so asking again doesn't help)
export function pairRandom( a: number, b: number ): number
{
    let h = ( Math.imul( a + 1, 73856093 ) ^ Math.imul( b + 1, 19349663 )) >>> 0;
    h = Math.imul( h ^ ( h >>> 16 ), 0x45d9f3b ) >>> 0;
    h = ( h ^ ( h >>> 16 )) >>> 0;
    return h / 4294967296;
}

// how well digestion d (0..1) handles a plant type; meat (type 2) is always 1
export function digestionEfficiency( digestion: number, type: number ): number
{
    if (( type === 2 ) || !TWO_PLANT_TYPES ) return 1;
    const x = ( type === PLANT_TYPE_WARM ) ? digestion : 1 - digestion;
    return DIGESTION_MINIMUM + ( 1 - DIGESTION_MINIMUM ) * Math.pow( x, DIGESTION_CURVE );
}

// The ecology gene block (genes 120..155).

// Genes 0..111 build the body. The ecological traits live in their own block

// unlike the body genes.

export function preyTargetScore( reward: number, pursuitCost: number, rewardWeight: number,
                                 travelWeight: number, baitBias = 0, isBait = false ): number
{
    return reward * rewardWeight - pursuitCost * travelWeight + ( isBait ? baitBias : -baitBias );
}

export const ECOLOGY_GENES_FIRST   = 120;
export const ECOLOGY_GENES_LAST    = 143;
export const MARKER_GENES_FIRST    = 144;                      // neutral markers: change slowly, affect nothing,
export const MARKER_GENES_LAST     = 159;                      // used to measure how related populations are

export const THERMAL_WIDTH_MIN     = 0.04;
export const THERMAL_WIDTH_MAX     = 0.40;

const ECOLOGY_MUTATION_RATE = 0.02;     // per gene per birth
const ECOLOGY_MUTATION_STEP = 9;        // typical size (out of 255)
const MARKER_MUTATION_RATE  = 0.0015;

export function isEcologyGene( g ) { return ( g >= ECOLOGY_GENES_FIRST ) && ( g <= ECOLOGY_GENES_LAST ); }
export function isMarkerGene ( g ) { return ( g >= MARKER_GENES_FIRST  ) && ( g <= MARKER_GENES_LAST  ); }

//------------------------------------------------------------
// a trait = the average of its genes, scaled to 0..1
//------------------------------------------------------------
export function geneAverage( genotype, genes: number[] ): number
{
    let sum = 0;
    for ( const g of genes ) { sum += genotype.getGeneValue(g); }
    return sum / ( genes.length * 255 );
}

// write a 0..1 value into all genes of a trait (optionally with a little noise per gene)
export function setTrait( genotype, genes: number[], value: number, noise = 0 )
{
    for ( const g of genes )
    {
        let v = value * 255 + ( noise > 0 ? gaussian() * noise * 255 : 0 );
        genotype.setGeneValue( g, clampGene( v ) );
    }
}

export function thermalWidthOf( genotype ): number
{
    return THERMAL_WIDTH_MIN + geneAverage( genotype, GENES_THERMAL_WIDTH ) * ( THERMAL_WIDTH_MAX - THERMAL_WIDTH_MIN );
}

export function thermalWidthToGene( w: number ): number
{
    return ( w - THERMAL_WIDTH_MIN ) / ( THERMAL_WIDTH_MAX - THERMAL_WIDTH_MIN );
}

//------------------------------------------------------------
// Saved pools from before stage 5 kept the predator level in gene 51 and the
// hunger in gene 50. Move them into this block; the ecology traits get
// neutral values (likes 0.5, broad tolerance, digests both plants).
//------------------------------------------------------------
export const GENE_LAYOUT_VERSION = 2;

export function convertOldGenes( genes: number[] ): void
{
    const predator = genes[51], hunger = genes[50];
    for ( const g of GENES_PREDATOR        ) { genes[g] = predator; }
    for ( const g of GENES_HUNGER          ) { genes[g] = hunger;   }
    for ( const g of GENES_THERMAL_OPTIMUM ) { genes[g] = 128;      }
    for ( const g of GENES_THERMAL_WIDTH   ) { genes[g] = 255;      }
    for ( const g of GENES_DIGESTION       ) { genes[g] = 128;      }
    for ( const g of GENES_BRIGHTNESS      ) { genes[g] = 128;      }
    for ( const g of GENES_CHOOSINESS      ) { genes[g] = 0;        }
}

//------------------------------------------------------------
// mutation of one gene of this block (returns the new value)
//------------------------------------------------------------
export function mutateEcologyGene( value: number ): number
{
    return clampGene( value + gaussian() * ECOLOGY_MUTATION_STEP );
}

export function mutateMarkerGene( value: number ): number
{
    return clampGene( value + ( gpRandom() < 0.5 ? -1 : 1 ) * ( gpRandom() < 0.5 ? 1 : 2 ) );
}

export function ecologyMutationRate( g ): number
{
    return isMarkerGene( g ) ? MARKER_MUTATION_RATE : ECOLOGY_MUTATION_RATE;
}

// whole number 0..255, reflected at the ends (no wrap-around)
function clampGene( v: number ): number
{
    if ( v < 0   ) { v = -v; }
    if ( v > 255 ) { v = 510 - v; }
    return Math.max( 0, Math.min( 255, Math.round( v )));
}

// standard normal random number
export function gaussian(): number
{
    let u = 0, v = 0;
    while ( u === 0 ) { u = gpRandom(); }
    while ( v === 0 ) { v = gpRandom(); }
    return Math.sqrt( -2.0 * Math.log( u )) * Math.cos( 2.0 * Math.PI * v );
}

 	}
	    
	//-------------------------------------------------------------
	// variables
	//-------------------------------------------------------------
    let _normalizedGenes        = new Array( NUM_GENES ); 
    let _geneNames              = new Array( NUM_GENES ); 
    let _branchStatus           = new Array( MAX_PARTS ); 
    let _categoryValues         = new Array( NUM_CATEGORIES ); 
    let _partIndex              = ZERO;
    let _generating             = false;
    let _frequency              = ZERO;
    let _numGenesUsed           = 0;
    let _numGenesPerCategory    = 0;

	this.getPreferredFoodTypeGene   = function() { return preferredFoodTypeGene;    }

    for (let g=0; g<NUM_GENES; g++)

        _geneNames[g] = "junk";
    }
    
         
	//----------------------------------------------------
	// generate phenotype from genotype
	//----------------------------------------------------
	this.generatePhenotypeFromGenotype = function( genotype )
	{
        //--------------------------------
        // create new phenotype...
        //--------------------------------
		let phenotype = new Phenotype();

	    // create categories array

		    _branchStatus[p] = false;
		}

	    //--------------------------------------------------------
	    // convert the gene values from byte to normalized
	    //--------------------------------------------------------
		for (let g=0; g<NUM_GENES; g++)
		{
            _normalizedGenes[g] = genotype.getGeneValue(g) / BYTE_SIZE;
		    assert( _normalizedGenes[g] >= ZERO, "normalizedGenes[g] >= ZERO" );

        //------------------------------------------------------------

        let splinedRange            = MAX_SPLINED           - MIN_SPLINED;
        let endCapSplineRange       = MAX_END_CAP_SPLINE    - MIN_END_CAP_SPLINE;
        phenotype.hungerThresholdRatio = 0.1 + geneAverage( genotype, GENES_HUNGER ) * 0.9;
        // the swimbot's own hunger threshold = the pool's setting x 0.5 .. 2
        // (gene 1/3 = exactly the setting); used in the environment pool
        phenotype.hungerFactor     = 0.5 + 1.5 * geneAverage( genotype, GENES_HUNGER );
        // autotomy (dropping a part of the body)
        phenotype.autotomyBite     = geneAverage( genotype, GENES_AUTOTOMY_BITE );                // chance when bitten
        phenotype.autotomyThreat   = geneAverage( genotype, GENES_AUTOTOMY_THREAT );              // 0 never .. 1 at first sight
        phenotype.autotomyAmount   = 0.1 + 0.5 * geneAverage( genotype, GENES_AUTOTOMY_AMOUNT );  // 10 .. 60 % of the parts
        phenotype.autotomyWhich    = geneAverage( genotype, GENES_AUTOTOMY_WHICH );


        // predator level and hunger live in the ecology gene block (EcologyGenes.ts)
        phenotype.predatorLevel    = geneAverage( genotype, GENES_PREDATOR ); // 0..1
        phenotype.thermalOptimum   = geneAverage( genotype, GENES_THERMAL_OPTIMUM );
        phenotype.thermalWidth     = thermalWidthOf( genotype );
        phenotype.digestion        = geneAverage( genotype, GENES_DIGESTION );
        // the body brightness others see: half its own genes, half the preferred
        // temperature (cold-adapted swimbots are darker, like real animals).
        // So the climate a swimbot is adapted to shows in its colors, and
        // choosing a mate of similar color means choosing one from a similar
        // climate (a "magic trait", see docs/БИОЛОГ_ЭТАП5.md)
        phenotype.brightness       = BRIGHTNESS_FROM_GENES * geneAverage( genotype, GENES_BRIGHTNESS )
                                   + ( 1 - BRIGHTNESS_FROM_GENES ) * phenotype.thermalOptimum;
        phenotype.choosiness       = geneAverage( genotype, GENES_CHOOSINESS );

        // Legacy/binary compatibility for other code that may still look for isPredator
        phenotype.isPredator       = phenotype.predatorLevel > PREDATOR_LEVEL_THRESHOLD;

        //---------------------------------
        // apply genes
        //---------------------------------
		let g = -1;
        
        g++; _frequency = MIN_FREQUENCY + frequencyRange    * _normalizedGenes[g];  _geneNames[g] = "frequency";
        g++; _cutOff    = MIN_CUT_OFF   + cutOffRange       * _normalizedGenes[g];  _geneNames[g] = "cutoff";
        
		for (let c=0; c<NUM_CATEGORIES; c++)
		{
		    _numGenesPerCategory = 0;
            g++; _categoryValues[c].startRed        = MIN_COLOR             + colorRange            * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "start red";
            g++; _categoryValues[c].startGreen      = MIN_COLOR             + colorRange            * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "start green";
            g++; _categoryValues[c].startBlue	    = MIN_COLOR             + colorRange            * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "start blue";
            g++; _categoryValues[c].endRed	        = MIN_COLOR             + colorRange            * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "end red";
            g++; _categoryValues[c].endGreen        = MIN_COLOR             + colorRange            * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "end green";
            g++; _categoryValues[c].endBlue         = MIN_COLOR             + colorRange            * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "end blue";
            g++; _categoryValues[c].startWidth      = MIN_WIDTH             + widthRange            * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "start width";
            g++; _categoryValues[c].endWidth        = MIN_WIDTH             + widthRange            * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "end width";
            g++; _categoryValues[c].startLength     = MIN_LENGTH            + lengthRange           * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "start length";
            g++; _categoryValues[c].endLength       = MIN_LENGTH            + lengthRange           * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "end length";


            g++; _categoryValues[c].amp             = MIN_AMP               + ampRange              * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "amplitude";
            g++; _categoryValues[c].phase           = MIN_PHASE             + phaseRange            * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "phase";
            g++; _categoryValues[c].turnAmp         = MIN_AMP               + ampRange              * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "turn amplitude";
            g++; _categoryValues[c].turnPhase       = MIN_PHASE             + phaseRange            * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "turn phase";
            g++; _categoryValues[c].branchAmp       = MIN_AMP               + ampRange              * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "branch amplitude";
            g++; _categoryValues[c].branchPhase     = MIN_PHASE             + phaseRange            * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "branch phase";
            g++; _categoryValues[c].branchTurnAmp   = MIN_AMP               + ampRange              * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "branch turn amplitude";
            g++; _categoryValues[c].branchTurnPhase = MIN_PHASE             + phaseRange            * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "branch turn phase";

            g++; _categoryValues[c].sequenceCount   = MIN_SEQUENCE_COUNT    + sequenceCountRange    * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "sequence count";
            g++; _categoryValues[c].branchPeriod    = MIN_BRANCH_PERIOD     + periodRange           * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "branch period";
            g++; _categoryValues[c].branchAngle     = MIN_BRANCH_ANGLE      + branchAngleRange      * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "branch angle";
            g++; _categoryValues[c].branchNumber    = MIN_BRANCH_NUMBER     + branchNumberRange     * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "branch number";
            g++; _categoryValues[c].branchShift     = MIN_BRANCH_SHIFT      + branchShiftRange      * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "branch shift";
            g++; _categoryValues[c].branchCategory  = MIN_BRANCH_CATEGORY   + branchCategoryRange   * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "branch category";
            g++; _categoryValues[c].branchReflect   = MIN_BRANCH_REFLECT    + branchReflectRange    * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "branch reflect";
            
            g++; _categoryValues[c].splined         = MIN_SPLINED           + splinedRange          * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "splined";
            g++; _categoryValues[c].endCapSpline    = MIN_END_CAP_SPLINE    + endCapSplineRange     * _normalizedGenes[g]; _numGenesPerCategory ++;  _geneNames[g] = "end cap spline";
         
            //---------------------------------------------------------------------------------------------
            // make these integers
            //---------------------------------------------------------------------------------------------
            _categoryValues[c].sequenceCount    = Math.floor( ZERO  + _categoryValues[c].sequenceCount  );
            _categoryValues[c].branchPeriod     = Math.floor( ZERO  + _categoryValues[c].branchPeriod   );
            _categoryValues[c].branchNumber     = Math.floor( ONE   + _categoryValues[c].branchNumber   );
            _categoryValues[c].branchShift      = Math.floor( ZERO  + _categoryValues[c].branchShift    );
            _categoryValues[c].branchCategory   = Math.floor( ZERO  + _categoryValues[c].branchCategory );
            _categoryValues[c].branchReflect    = Math.floor( ONE   + _categoryValues[c].branchReflect  );
            _categoryValues[c].splined          = Math.round( ZERO  + _categoryValues[c].splined        );
        }

        //-----------------------------------------------------------------------------------------
        // add genes for food type preference and digestibility:
        //
        // by default, swimbots are all born with a preferrence for 0 (green), but if numFoodTypes 
        // is set to 2, then they are born with a genetically-determined preferrence.
        //-----------------------------------------------------------------------------------------
        phenotype.preferredFoodType  = 0;
        phenotype.digestibleFoodType = 0;


        g++;  
        preferredFoodTypeGene = g; 
        _geneNames[g] = "preferred food type";
        if ( globalTweakers.numFoodTypes === 2 ) 
        {
            phenotype.preferredFoodType = Math.floor( _normalizedGenes[g] * 2 );     
        }   

        g++;  
        digestibleFoodTypeGene = g; 
        _geneNames[g] = "digestible food type";
        if ( globalTweakers.numFoodTypes === 2 ) 
        {
            phenotype.digestibleFoodType = Math.floor( _normalizedGenes[g] * 2 );     
        }   
        

        //---------------------------------------
        // important: set _numGenesUsed
        //---------------------------------------
        _numGenesUsed = g + 1;

        //---------------------------------------
        // make sure this is kosher
        //---------------------------------------
        //console.log( "num genes used = " + _numGenesUsed + " out of " + NUM_GENES );
		assert( _numGenesUsed < NUM_GENES, "embryology: _numGenesUsed < NUM_GENES" );
        
        //---------------------------------
        // set the frequency...
        //---------------------------------
        phenotype.frequency = _frequency;
        
        // ВРЕМЕННО: хищники — чёрные, чтобы их было видно
        if (phenotype.predatorLevel > PREDATOR_LEVEL_THRESHOLD) {
            for (let c = 0; c < NUM_CATEGORIES; c++) {
                _categoryValues[c].startRed   = 0;
                _categoryValues[c].startGreen = 0;
                _categoryValues[c].startBlue  = 0;
                _categoryValues[c].endRed     = 0;
                _categoryValues[c].endGreen   = 0;
                _categoryValues[c].endBlue    = 0;
            }
        }
        //------------------------------------------------------------------
        // environment: the body brightness gene shows in the colors (the
        // colors move towards a grey of that brightness), so swimbots that
        // choose a mate of similar color also choose similar brightness
        //------------------------------------------------------------------
        else if ( environment.active )
        {
            const grey = phenotype.brightness;
            const k    = BRIGHTNESS_COLOR_SHARE;
            for (let c = 0; c < NUM_CATEGORIES; c++)
            {
                const v = _categoryValues[c];
                v.startRed   = v.startRed   * ( 1 - k ) + grey * k;
                v.startGreen = v.startGreen * ( 1 - k ) + grey * k;
                v.startBlue  = v.startBlue  * ( 1 - k ) + grey * k;
                v.endRed     = v.endRed     * ( 1 - k ) + grey * k;
                v.endGreen   = v.endGreen   * ( 1 - k ) + grey * k;
                v.endBlue    = v.endBlue    * ( 1 - k ) + grey * k;
            }
        }

        //----------------------------------------------
        // generate the first sequence...
        //----------------------------------------------
        _partIndex = ROOT_PART;
        let startCategory = 0;
        
testNoEel = true;
//console.log( "--------------");
        this.generateBodySequence( phenotype, _partIndex, ZERO, startCategory, ONE );  
testNoEel = false;  


        //----------------------------------------------
        // generate the rest of the body...
        //----------------------------------------------
        _generating = true;
        while ( _generating )
        {          
            for (let p=0; p<MAX_PARTS; p++)
            {
                _generating = false; // this might get set back to true in generateBodySequence
                
                //----------------------------------------------
                // branching...
                //----------------------------------------------
                if ( _branchStatus[p] )
                {        
                    _branchStatus[p] = false; // this might get set back to true in generateBodySequence
                    
                    let partCategory = phenotype.parts[p].category;              
                    
                    let c = _categoryValues[ partCategory ].branchCategory;
                    let reflect = ONE;

                    //--------------------------------------------
                    // grow branch 
                    //--------------------------------------------
                    if ( _categoryValues[c].branchNumber === 1 )
                    {
                        reflect = ONE; 
                        this.generateBodySequence( phenotype, p, _categoryValues[c].branchAngle, c, reflect );   
                    }
                    else
                    {
                        //---------------------------------------------------------------
                        // fan out branch angle across the range of branches....
                        //---------------------------------------------------------------
                        for (let b=0; b<_categoryValues[c].branchNumber; b++)
                        {
                            reflect = ONE; 
                            if ( b % _categoryValues[c].branchReflect === 0 )
                            {
                                reflect = -ONE;
                            }
                            
                            let f = -ONE + ( b / ( _categoryValues[c].branchNumber - 1 ) ) * 2;

                            this.generateBodySequence( phenotype, p, _categoryValues[c].branchAngle * f, c, reflect );    
                        }   
                    }                    
                }
            }
        }
        
        //------------------------------------------------------------------------
        // set num parts (it will have accumulated from generating part sequences)
        //------------------------------------------------------------------------
        phenotype.numParts = _partIndex + 1;
        
        assert( phenotype.numParts > 1, "phenotype.numParts > 1"  );

		//-----------------------------------------------------
		// re-order the parts for more sensible rendering 
		//-----------------------------------------------------
//this.fixPartOrdering( phenotype );

		//----------------------
		// return phenotype
		//----------------------
        return phenotype;
    }

	//---------------------------------------------------------------------------------

	this.generateBodySequence = function( phenotype, parent, branchAngle, c, reflect )
	{
        for (let i=0; i<_categoryValues[c].sequenceCount; i++)
        {
            if ( _partIndex < _cutOff )
            {        
                //-------------------------
                // increment _partIndex  
                //-------------------------
                _partIndex ++;               
                assert( _partIndex < MAX_PARTS, "_partIndex < MAX_PARTS" );
                
                phenotype.parts[ _partIndex ].child = NULL_INDEX; //default
            
                //-------------------------------------------------------
                // the first part is a branchpoint from the parent  
                //-------------------------------------------------------
                if ( i === 0 )
                {
                    phenotype.parts[ _partIndex ].branch    = true;
                    phenotype.parts[ _partIndex ].parent    = parent;
                    phenotype.parts[ _partIndex ].angle     = branchAngle; 
                    phenotype.parts[ _partIndex ].amp       = _categoryValues[c].branchAmp;
                    phenotype.parts[ _partIndex ].phase     = _categoryValues[c].branchPhase * _partIndex; 
                    phenotype.parts[ _partIndex ].turnAmp   = _categoryValues[c].branchTurnAmp;
                    phenotype.parts[ _partIndex ].turnPhase = _categoryValues[c].branchTurnPhase * _partIndex;   
                }
                else
                {
                    let parent = _partIndex - 1;
                    phenotype.parts[ parent ].child = _partIndex;
                
                    phenotype.parts[ _partIndex ].branch    = false;
                    phenotype.parts[ _partIndex ].parent    = parent;
                    phenotype.parts[ _partIndex ].angle     = ZERO;

                    phenotype.parts[ _partIndex ].amp       = _categoryValues[c].amp;
                    phenotype.parts[ _partIndex ].phase     = _categoryValues[c].phase * _partIndex; 
                    phenotype.parts[ _partIndex ].turnAmp   = _categoryValues[c].turnAmp;
                    phenotype.parts[ _partIndex ].turnPhase = _categoryValues[c].turnPhase;   
                }


                
if ( testNoEel )
{
    //console.log( "testNoEel" );
    phenotype.parts[ _partIndex ].turnAmp   = ZERO;
    phenotype.parts[ _partIndex ].turnPhase = ZERO;   
}
        
                //-----------------------------------------------
                // apply reflection on amp
                //-----------------------------------------------
                phenotype.parts[ _partIndex ].amp *= reflect;
                
                //---------------------------------------------------
                // set some other attributes  
                //---------------------------------------------------
                phenotype.parts[ _partIndex ].category      = c;
                phenotype.parts[ _partIndex ].frequency     = phenotype.frequency;
                phenotype.parts[ _partIndex ].splined       = _categoryValues[c].splined;
                phenotype.parts[ _partIndex ].endCapSpline  = _categoryValues[c].endCapSpline;
 
                //----------------------------------------------------
                // set attributes that interpolate over the sequence
                //----------------------------------------------------
                let fraction = ZERO;
                
                if ( _categoryValues[c].sequenceCount > 1 )
                {
                    fraction = i / ( _categoryValues[c].sequenceCount - 1 );     
                }
                        
                phenotype.parts[ _partIndex ].width  = _categoryValues[c].startWidth  + fraction * ( _categoryValues[c].endWidth  - _categoryValues[c].startWidth   );
                phenotype.parts[ _partIndex ].length = _categoryValues[c].startLength + fraction * ( _categoryValues[c].endLength - _categoryValues[c].startLength  );   
                phenotype.parts[ _partIndex ].red    = _categoryValues[c].startRed    + fraction * ( _categoryValues[c].endRed    - _categoryValues[c].startRed     );
                phenotype.parts[ _partIndex ].green  = _categoryValues[c].startGreen  + fraction * ( _categoryValues[c].endGreen  - _categoryValues[c].startGreen   );
                phenotype.parts[ _partIndex ].blue   = _categoryValues[c].startBlue   + fraction * ( _categoryValues[c].endBlue   - _categoryValues[c].startBlue    );
                
			    assert( phenotype.parts[ _partIndex ].length > ZERO, "In Embryology: phenotype.parts[ _partIndex ].length > ZERO" );
			    assert( phenotype.parts[ _partIndex ].width  > ZERO, "In Embryology: phenotype.parts[ _partIndex ].width  > ZERO" );              

                //---------------------------------------------------------------------------------
                // determine if there is a branching
                //---------------------------------------------------------------------------------
                let mod = ( i + _categoryValues[c].branchShift ) % _categoryValues[c].branchPeriod;
               
                if ( mod === 0 )
                {
                    _generating = true;
        		    _branchStatus[ _partIndex ] = true;
                }
            }   
        }
    }

//------------------------------------------------------------------------------
// The environment: a temperature map with biomes, seasons, and a count of the
// plants in each part of the pool.
//
// Temperature T (0 = cold .. 1 = hot) =
//     0.5 + 0.30 · latitude (north cold, south hot)
//         + 0.20 · biome offset (patches 2500–4000 units, soft 400–800 borders)
//         + season (period 10 000 steps, stronger towards the poles)
//
// Background brightness B (0 = dark bottom .. 1 = light bottom): half the
// temperature, half a random shade of each biome patch. Swimbots whose body
// brightness matches the background are harder for predators to see.
//
// Why these numbers: see docs/БИОЛОГ_ЭТАП5.md (patches must be much larger than
// how far a swimbot travels in its life, borders soft, changes slow).
//------------------------------------------------------------------------------
import { gpRandom } from "./Globals";
import { POOL_BOTTOM, POOL_LEFT, POOL_RIGHT, POOL_TOP } from "./Pool";
import { petriLevel, petriPosition, petriTemperature, petriSettings } from "./Petri";

// Control experiment "no environment": set to true and the map is flat
// (T = 0.5 everywhere, plants grow anywhere) while all other rules stay.
export const FLAT_ENVIRONMENT          = false;

// Optional matter cycle experiment: meat rot returns nutrients to the same
// plant cell, and each successful sprout consumes one nutrient unit.
export const NUTRIENT_CYCLE_ENABLED              = false;
export const INITIAL_SOIL_NUTRIENT_PER_CELL      = 3.0;
export const SOIL_NUTRIENT_CAPACITY_PER_CELL     = 12.0;
export const SOIL_NUTRIENT_PER_SPROUT            = 1.0;
export const MEAT_DETRITUS_YIELD                 = 0.05;
export const SOIL_NUTRIENT_DECAY_PER_STEP        = 0.00001;

// Optional seasonal availability pulse for the single plant food type.
// The multiplier averages to 1 over a full season and is disabled by default.
export const SEASONAL_PLANT_GROWTH_ENABLED       = false;
export const SEASONAL_PLANT_GROWTH_AMPLITUDE     = 0.5;

export const FIELD_SIZE                = 256;      // temperature grid: FIELD_SIZE x FIELD_SIZE cells
export const PLANT_CELL_SIZE           = 250;      // plant counting grid (pool units)

const BIOME_STRENGTH                   = 0.20;
const BIOME_SPACING                    = 4000;     // distance between biome patch centers
const BIOME_BORDER_SOFTNESS            = 300;      // larger = wider, softer borders
const BIOME_OFFSETS                    = [ -1.0, -0.5, 0.0, 0.5, 1.0 ];   // 5 biome types

export const SEASON_PERIOD             = 10000;    // steps
export const SEASON_AMPLITUDE_EQUATOR         = 0.02;
export const SEASON_AMPLITUDE_POLES           = 0.08;

const PLANT_OPTIMUM                    = [ 0.7, 0.3 ];
const PLANT_TOLERANCE                  = 0.15;

export class Environment
{
    private baseTemperature = new Float32Array( FIELD_SIZE * FIELD_SIZE );
    private biomeOfCell     = new Int8Array   ( FIELD_SIZE * FIELD_SIZE );
    private background      = new Float32Array( FIELD_SIZE * FIELD_SIZE );
    private biomeSeeds      : { x: number, y: number, type: number, shade: number }[] = [];
    private time            = 0;

    // the environment rules (temperature, climate plants, digestion) apply
    // only in the "random" pool; the classic presets keep the old rules
    active                  = false;

    // the "petri" pool: zones instead of biome patches (null = normal map)
    petri : { zones: number, layout: number, softness: number } | null = null;
    private zoneOfCell      = new Int8Array( FIELD_SIZE * FIELD_SIZE );

    readonly plantColumns   = Math.ceil( ( POOL_RIGHT  - POOL_LEFT ) / PLANT_CELL_SIZE );
    readonly plantRows      = Math.ceil( ( POOL_BOTTOM - POOL_TOP  ) / PLANT_CELL_SIZE );
    readonly plantCount     = [ new Int32Array( this.plantColumns * this.plantRows ),     // warm plants per cell
                                new Int32Array( this.plantColumns * this.plantRows ) ];   // cold plants per cell
    private soilNutrient     = new Float32Array( this.plantColumns * this.plantRows );
    private nutrientUpdatedAt = new Int32Array( this.plantColumns * this.plantRows );

    //--------------------------------------------------------------------
    // make a new map (uses the simulation's random numbers)
    //--------------------------------------------------------------------
    generate( petri = false ): void
    {
        this.petri = petri ? { zones: petriSettings.zones, layout: petriSettings.layout, softness: petriSettings.softness } : null;
        if ( this.petri ) { this.biomeSeeds = []; this.computePetriField(); return; }

        //------------------------------------------------------------
        // biome patch centers on a jittered grid, random biome types
        //------------------------------------------------------------
        this.biomeSeeds = [];
        const width  = POOL_RIGHT  - POOL_LEFT;
        const height = POOL_BOTTOM - POOL_TOP;
        const columns = Math.max( 1, Math.round( width  / BIOME_SPACING ));
        const rows    = Math.max( 1, Math.round( height / BIOME_SPACING ));
        for ( let r = 0; r < rows; r++ )
        {
            for ( let c = 0; c < columns; c++ )
            {
                this.biomeSeeds.push(
                {
                    x    : POOL_LEFT + ( c + 0.2 + 0.6 * gpRandom() ) * width  / columns,
                    y    : POOL_TOP  + ( r + 0.2 + 0.6 * gpRandom() ) * height / rows,
                    type : Math.floor( gpRandom() * BIOME_OFFSETS.length ),
                    shade: gpRandom(),
                });
            }
        }

        this.computeField();
    }

    //--------------------------------------------------------------------
    // saving and loading: the map is rebuilt from its patch centers
    //--------------------------------------------------------------------
    getState()
    {
        return { active: this.active, petri: this.petri, seeds: this.biomeSeeds.map( s => ({ x: s.x, y: s.y, type: s.type, shade: s.shade })) };
    }

    // petri settings from a file: whole numbers in range, or the defaults
    static checkedPetri( p ): { zones: number, layout: number, softness: number }
    {
        const zones    = Math.round( Number( p.zones ));
        const layout   = Math.round( Number( p.layout ));
        const softness = Number( p.softness );
        return {
            zones   : Number.isFinite( zones )    ? Math.min( 12, Math.max( 2, zones )) : 7,
            layout  : Number.isFinite( layout ) && layout >= 0 ? layout : 0,
            softness: Number.isFinite( softness ) ? Math.min( 1, Math.max( 0, softness )) : 0.1,
        };
    }

    setState( state ): void
    {
        this.petri = ( state && state.petri ) ? Environment.checkedPetri( state.petri ) : null;
        if ( state && state.active && this.petri )
        {
            // the panel shows (and a new petri pool uses) the loaded settings
            petriSettings.zones    = this.petri.zones;
            petriSettings.layout   = this.petri.layout;
            petriSettings.softness = this.petri.softness;
            this.active = true; this.biomeSeeds = []; this.computePetriField(); return;
        }
        this.active     = !!( state && state.active && state.seeds && state.seeds.length > 0 );
        this.biomeSeeds = this.active ? state.seeds.map( s => ({ x: s.x, y: s.y, type: s.type, shade: ( s.shade !== undefined ) ? s.shade : 0.5 })) : [];
        if ( this.active ) { this.computeField(); }
    }

    //--------------------------------------------------------------------
    // the "petri" pool: temperature by zone (see Petri.ts), no seasons
    //--------------------------------------------------------------------
    private computePetriField(): void
    {
        const p = this.petri;
        for ( let j = 0; j < FIELD_SIZE; j++ )
        {
            for ( let i = 0; i < FIELD_SIZE; i++ )
            {
                const level = petriLevel( petriPosition( ( i + 0.5 ) / FIELD_SIZE, ( j + 0.5 ) / FIELD_SIZE, p.layout ), p.zones, p.softness );
                const t = petriTemperature( level, p.zones );
                const k = j * FIELD_SIZE + i;
                this.baseTemperature[k] = t;
                this.background     [k] = 0.25 + 0.5 * t;
                this.zoneOfCell     [k] = Math.round( level );
                this.biomeOfCell    [k] = 0;
            }
        }
    }

    // which petri zone a point is in (0 .. zones-1)
    zoneAt( x: number, y: number ): number
    {
        const i = Math.min( FIELD_SIZE - 1, Math.max( 0, Math.floor( ( x - POOL_LEFT ) / ( POOL_RIGHT  - POOL_LEFT ) * FIELD_SIZE )));
        const j = Math.min( FIELD_SIZE - 1, Math.max( 0, Math.floor( ( y - POOL_TOP  ) / ( POOL_BOTTOM - POOL_TOP  ) * FIELD_SIZE )));
        return this.zoneOfCell[ j * FIELD_SIZE + i ];
    }

    //--------------------------------------------------------------------
    // base temperature in every cell, from the biome patch centers
    //--------------------------------------------------------------------
    private computeField(): void
    {
        const width  = POOL_RIGHT  - POOL_LEFT;
        const height = POOL_BOTTOM - POOL_TOP;
        for ( let j = 0; j < FIELD_SIZE; j++ )
        {
            for ( let i = 0; i < FIELD_SIZE; i++ )
            {
                const x = POOL_LEFT + ( i + 0.5 ) / FIELD_SIZE * width;
                const y = POOL_TOP  + ( j + 0.5 ) / FIELD_SIZE * height;
                const latitude = this.latitude( y );

                // soft Voronoi: nearby patch centers blend near the borders
                let nearest = Infinity, nearestType = 0;
                const distances = this.biomeSeeds.map( s => Math.hypot( s.x - x, s.y - y ));
                for ( let k = 0; k < distances.length; k++ )
                {
                    if ( distances[k] < nearest ) { nearest = distances[k]; nearestType = this.biomeSeeds[k].type; }
                }
                let weightSum = 0, offsetSum = 0, shadeSum = 0;
                for ( let k = 0; k < distances.length; k++ )
                {
                    const w = Math.exp( -( distances[k] - nearest ) / BIOME_BORDER_SOFTNESS );
                    weightSum += w;
                    offsetSum += w * BIOME_OFFSETS[ this.biomeSeeds[k].type ];
                    shadeSum  += w * this.biomeSeeds[k].shade;
                }

                const t = Math.min( 1, Math.max( 0, 0.5 + LATITUDE_STRENGTH * latitude + BIOME_STRENGTH * ( offsetSum / weightSum )));
                this.baseTemperature[ j * FIELD_SIZE + i ] = t;
                this.background     [ j * FIELD_SIZE + i ] = 0.5 * t + 0.5 * ( shadeSum / weightSum );
                this.biomeOfCell    [ j * FIELD_SIZE + i ] = nearestType;
            }
        }
    }

    // -1 at the top (north, cold) .. +1 at the bottom (south, hot)
    latitude( y: number ): number
    {
        return ( ( y - POOL_TOP ) / ( POOL_BOTTOM - POOL_TOP ) ) * 2 - 1;
    }

    setTime( step: number ): void { this.time = step; }

    resetSoilNutrients( startingAmount: number ): void
    {
        const amount = Math.max( 0, Math.min( SOIL_NUTRIENT_CAPACITY_PER_CELL, startingAmount ));
        this.soilNutrient.fill( amount );
        this.nutrientUpdatedAt.fill( this.time );
    }

    private ageSoilNutrient( cell: number ): void
    {
        const elapsed = this.time - this.nutrientUpdatedAt[ cell ];
        if ( elapsed <= 0 ) return;
        if ( SOIL_NUTRIENT_DECAY_PER_STEP > 0 )
        {
            this.soilNutrient[ cell ] *= Math.exp( -SOIL_NUTRIENT_DECAY_PER_STEP * elapsed );
        }
        this.nutrientUpdatedAt[ cell ] = this.time;
    }

    returnMeatDetritusAt( x: number, y: number, meatEnergy: number ): number
    {
        if ( !NUTRIENT_CYCLE_ENABLED || !( meatEnergy > 0 )) return 0;
        const cell = this.plantCellIndex( x, y );
        this.ageSoilNutrient( cell );
        const returned = Math.min(
            Math.max( 0, SOIL_NUTRIENT_CAPACITY_PER_CELL - this.soilNutrient[ cell ] ),
            meatEnergy * MEAT_DETRITUS_YIELD
        );
        this.soilNutrient[ cell ] += returned;
        return returned;
    }

    takeSoilNutrientForSproutAt( x: number, y: number ): boolean
    {
        if ( !NUTRIENT_CYCLE_ENABLED ) return true;
        const cell = this.plantCellIndex( x, y );
        this.ageSoilNutrient( cell );
        if ( this.soilNutrient[ cell ] < SOIL_NUTRIENT_PER_SPROUT ) return false;
        this.soilNutrient[ cell ] -= SOIL_NUTRIENT_PER_SPROUT;
        return true;
    }

    getSoilNutrientAtCell( cell: number ): number
    {
        if ( cell < 0 || cell >= this.soilNutrient.length ) return 0;
        this.ageSoilNutrient( cell );
        return this.soilNutrient[ cell ];
    }

    // -1 (deep winter) .. +1 (high summer)
    seasonPhase(): number { return Math.sin( 2 * Math.PI * this.time / SEASON_PERIOD ); }

    // the season now (the warm part of the year is "summer")
    seasonName(): string
    {
        const phase = 2 * Math.PI * this.time / SEASON_PERIOD;
        const s = Math.sin( phase ), c = Math.cos( phase );
        if ( s >  0.5 ) return "summer";
        if ( s < -0.5 ) return "winter";
        return ( c > 0 ) ? "spring" : "autumn";
    }

    //--------------------------------------------------------------------
    // temperature at a point, now (smoothly interpolated, with season)
    //--------------------------------------------------------------------
    temperatureAt( x: number, y: number ): number
    {
        if ( !this.active || FLAT_ENVIRONMENT ) return 0.5;

        const fx = ( x - POOL_LEFT ) / ( POOL_RIGHT  - POOL_LEFT ) * FIELD_SIZE - 0.5;
        const fy = ( y - POOL_TOP  ) / ( POOL_BOTTOM - POOL_TOP  ) * FIELD_SIZE - 0.5;
        let i = Math.floor( fx ), j = Math.floor( fy );
        const ax = fx - i, ay = fy - j;
        const i0 = Math.min( FIELD_SIZE - 1, Math.max( 0, i     ));
        const i1 = Math.min( FIELD_SIZE - 1, Math.max( 0, i + 1 ));
        const j0 = Math.min( FIELD_SIZE - 1, Math.max( 0, j     ));
        const j1 = Math.min( FIELD_SIZE - 1, Math.max( 0, j + 1 ));
        const f = this.baseTemperature;
        const base = ( f[ j0 * FIELD_SIZE + i0 ] * ( 1 - ax ) + f[ j0 * FIELD_SIZE + i1 ] * ax ) * ( 1 - ay )
                   + ( f[ j1 * FIELD_SIZE + i0 ] * ( 1 - ax ) + f[ j1 * FIELD_SIZE + i1 ] * ax ) * ay;

        const latitude  = Math.abs( this.latitude( y ));
        const amplitude = this.petri ? 0 : SEASON_AMPLITUDE_EQUATOR + ( SEASON_AMPLITUDE_POLES - SEASON_AMPLITUDE_EQUATOR ) * latitude;
        const t = base + amplitude * Math.sin( 2 * Math.PI * this.time / SEASON_PERIOD );
        return t < 0 ? 0 : ( t > 1 ? 1 : t );
    }

    // the clock and the temperature map (for the decision kernel's threads)
    getTime(): number { return this.time; }
    copyTemperatureTo( target: Float32Array ): void { target.set( this.baseTemperature ); }

    // copy the bottom brightness map (for the perception threads)
    copyBackgroundTo( target: Float32Array ): void { target.set( this.background ); }

    // background brightness at a point (0 = dark .. 1 = light)
    backgroundAt( x: number, y: number ): number
    {
        if ( !this.active || FLAT_ENVIRONMENT ) return 0.5;
        const i = Math.min( FIELD_SIZE - 1, Math.max( 0, Math.floor( ( x - POOL_LEFT ) / ( POOL_RIGHT  - POOL_LEFT ) * FIELD_SIZE )));
        const j = Math.min( FIELD_SIZE - 1, Math.max( 0, Math.floor( ( y - POOL_TOP  ) / ( POOL_BOTTOM - POOL_TOP  ) * FIELD_SIZE )));
        return this.background[ j * FIELD_SIZE + i ];
    }

    // which biome type (0..4) a point belongs to
    biomeAt( x: number, y: number ): number
    {
        const i = Math.min( FIELD_SIZE - 1, Math.max( 0, Math.floor( ( x - POOL_LEFT ) / ( POOL_RIGHT  - POOL_LEFT ) * FIELD_SIZE )));
        const j = Math.min( FIELD_SIZE - 1, Math.max( 0, Math.floor( ( y - POOL_TOP  ) / ( POOL_BOTTOM - POOL_TOP  ) * FIELD_SIZE )));
        return this.biomeOfCell[ j * FIELD_SIZE + i ];
    }

    // how well a plant type grows at temperature T (0..1)
    plantSuitability( type: number, temperature: number ): number
    {
        if ( !this.active || FLAT_ENVIRONMENT ) return 1;
        const d = temperature - PLANT_OPTIMUM[ type ];
        return Math.exp( -( d * d ) / ( 2 * PLANT_TOLERANCE * PLANT_TOLERANCE ));
    }

    //--------------------------------------------------------------------
    // plant counts per cell (rebuilt by GenePool every step)
    //--------------------------------------------------------------------
    plantCellIndex( x: number, y: number ): number
    {
        const c = Math.min( this.plantColumns - 1, Math.max( 0, Math.floor( ( x - POOL_LEFT ) / PLANT_CELL_SIZE )));
        const r = Math.min( this.plantRows    - 1, Math.max( 0, Math.floor( ( y - POOL_TOP  ) / PLANT_CELL_SIZE )));
        return r * this.plantColumns + c;
    }

    clearPlantCounts(): void
    {
        this.plantCount[0].fill( 0 );
        this.plantCount[1].fill( 0 );
    }

    //--------------------------------------------------------------------
    // one center for a "garden" in each biome type (for the start)
    //--------------------------------------------------------------------
    gardenCenters(): { x: number, y: number, type: number }[]
    {
        const centers = [];
        const margin = 1500;
        for ( let type = 0; type < BIOME_OFFSETS.length; type++ )
        {
            const candidates = this.biomeSeeds.filter( s => s.type === type
                && s.x > POOL_LEFT + margin && s.x < POOL_RIGHT  - margin
                && s.y > POOL_TOP  + margin && s.y < POOL_BOTTOM - margin );
            const pool = candidates.length > 0 ? candidates : this.biomeSeeds.filter( s => s.type === type );
            if ( pool.length > 0 ) { centers.push( pool[ Math.floor( gpRandom() * pool.length ) ] ); }
        }
        return centers;
    }

    //--------------------------------------------------------------------
    // a picture of the temperature map (for drawing behind the pool);
    // returns null outside a browser
    //--------------------------------------------------------------------
    makeImage(): any
    {
        // a page canvas, or in the simulation thread an offscreen one
        let image: any = null;
        if ( typeof document !== "undefined" )          { image = document.createElement( "canvas" ); }
        else if ( typeof OffscreenCanvas !== "undefined" ) { image = new OffscreenCanvas( FIELD_SIZE, FIELD_SIZE ); }
        else return null;
        image.width = image.height = FIELD_SIZE;
        const context = image.getContext( "2d" );
        const pixels  = context.createImageData( FIELD_SIZE, FIELD_SIZE );
        for ( let k = 0; k < FIELD_SIZE * FIELD_SIZE; k++ )
        {
            // hue = temperature (cold: deep blue-grey; hot: muted brown-olive),
            // lightness = background brightness (the pool keeps its calm look)
            const on = this.active && !FLAT_ENVIRONMENT;
            const t  = on ? this.baseTemperature[k] : 0.5;
            const l  = 0.62 + 0.76 * ( on ? this.background[k] : 0.5 );
            pixels.data[ k * 4     ] = ( 38 + t * 40 ) * l;
            pixels.data[ k * 4 + 1 ] = ( 52 + t * 18 ) * l;
            pixels.data[ k * 4 + 2 ] = ( 82 - t * 34 ) * l;
            pixels.data[ k * 4 + 3 ] = 255;
        }
        context.putImageData( pixels, 0, 0 );
        return image;
    }
}

	this.setAsOffspring = function( parent_0, parent_1 )
	{
	    // the brain chromosome: whole neurons from either parent, small mutations
	    inheritBrain( _brain, parent_0.getBrain(), parent_1.getBrain(), SELF_CANNIBALISM_ENABLED,
	                  PREDATOR_TARGET_SELECTION_ENABLED );

	    //console.log( parent_0 );
	    //console.log( parent_1 );

	    /*
	    console.log( "----------------------------");
	    console.log( "setAsOffspring");
	    console.log( "----------------------------");

        for (let g=0; g<NUM_GENES; g++)
        {
            console.log( parent_0.genes[g] + ", " + parent_1.genes[g] );
        }
	    */

        //-------------------------------------------
        // start with random parent either 1 or 2
        //-------------------------------------------
        let parent = 0;
        if ( gpRandom() < ONE_HALF )
        {
            parent = 1;
        }

        //-------------------------------------------
        // scan genes
        //-------------------------------------------
        for (let g=0; g<NUM_GENES; g++ )
        {
            //-----------------------------------
            // crossover - switch to other parent
            //-----------------------------------
            if ( gpRandom() < CROSSOVER_RATE )
            {
                if ( parent === 0 )
                {
                    parent =  1;
                }
                else
                {
                    parent = 0;
                }
            }

            //-----------------------------------
            // copy parent gene to child gene
            //-----------------------------------
            if ( parent === 0 )
            {
                assert ( parent_0.getGeneValue(g) >= 0,         "Genotype: setAsOffspring: parent_0.getGeneValue(g) >= 0" );
                assert ( parent_0.getGeneValue(g) < BYTE_SIZE,  "Genotype: setAsOffspring: parent_0.getGeneValue(g) < BYTE_SIZE" );
                assertInteger( parent_0.getGeneValue(g),        "Genotype: setAsOffspring: assertInteger: parent_0.getGeneValue(g)" );

                _genes[g] = parent_0.getGeneValue(g);
            }
            else
            {
                assert ( parent_1.getGeneValue(g) >= 0,         "Genotype: setAsOffspring: parent_1.getGeneValue(g) >= 0" );
                assert ( parent_1.getGeneValue(g) < BYTE_SIZE,  "Genotype: setAsOffspring: parent_1.getGeneValue(g) < BYTE_SIZE" );
                assertInteger( parent_1.getGeneValue(g),        "Genotype: setAsOffspring: assertInteger: parent_1.getGeneValue(g)" );

                _genes[g] = parent_1.getGeneValue(g);
            }

            assertInteger( _genes[g], "Genotype: setAsOffspring: assertInteger: _genes[g]" );

            //-----------------------------------
            // mutation
            //-----------------------------------
            if (( SELF_CANNIBALISM_ENABLED && isSelfFeedingGene(g) )
            ||  isEcologyGene(g) || isMarkerGene(g) || isDeferenceGene(g) )
            {
                // ecology genes and markers: small steps, no wrap-around
                if ( gpRandom() < ecologyMutationRate(g) )
                {
                    _genes[g] = isMarkerGene(g) ? mutateMarkerGene( _genes[g] ) : mutateEcologyGene( _genes[g] );
                }
            }
            else if ( gpRandom() < MUTATION_RATE )
            {
                this.mutateGene(g);
            }

            assert ( _genes[g] >= 0, "_genes[g] >=   0" );
            assert ( _genes[g] < BYTE_SIZE, "_genes[g] < BYTE_SIZE" );
            assertInteger( _genes[g], "Genotype: setAsOffspring: AFTER MUTATION...assertInteger: _genes[g]" );
        }
    }

        let amplitude = Math.floor( gpRandom() * gpRandom() * BYTE_SIZE );
        //console.log( "amplitude = " + amplitude );

        //-------------------------------------
        // keep it an integer!!!
        //-------------------------------------
        amplitude = Math.round( amplitude );

        assert( amplitude >= 0, "mutateGene:amplitude >= 0" );
        assert( amplitude < BYTE_SIZE, "mutateGene:amplitude < BYTE_SIZE" );

        if ( gpRandom() > ONE_HALF )
        {
            let before = _genes[g];
            _genes[g] += amplitude;

            if ( _genes[g] >= BYTE_SIZE )
            {
                _genes[g] -= BYTE_SIZE;
            }

            //console.log( "gene " + g + " mutated by " + amplitude + "; the value changed from " + before + " to " + _genes[g] );
        }
        else
        {
            _genes[g] -= amplitude;

            if ( _genes[g] < 0 )
            {
                _genes[g] += BYTE_SIZE;
            }
        }


        assertInteger( _genes[g], "Genotype: mutateGene" );


        assert ( _genes[g] >= 0, "Genotype: mutateGene:_genes[g] >=   0" );
        assert ( _genes[g] < BYTE_SIZE, "Genotype: mutateGene:_genes[g] < BYTE_SIZE" );
    }


    //-----------------------------
	this.zap = function( amount )
	{
        for (let g=0; g<NUM_GENES; g++ )
        {
            if ( gpRandom() < amount )
            {
                if      ( isMarkerGene(g)  ) { _genes[g] = mutateMarkerGene ( _genes[g] ); }
                else if ( SELF_CANNIBALISM_ENABLED && isSelfFeedingGene(g) ) { _genes[g] = mutateEcologyGene( _genes[g] ); }
                else if ( isEcologyGene(g) || isDeferenceGene(g) ) { _genes[g] = mutateEcologyGene( _genes[g] ); }
                else                         { this.mutateGene(g); }
            }
        }
	    mutateBrain( _brain, amount, SELF_CANNIBALISM_ENABLED, PREDATOR_TARGET_SELECTION_ENABLED );
    }
}

// Neural brain (stage 6).
//
// A small fixed network decides WHAT a swimbot wants (eat, mate, flee, hunt)
// and HOW it moves (follow the climate, keep close to or away from others,
// how hard to swim). Its weights are the "brain chromosome": inherited from
// both parents (whole neurons at a time), with small random mutations.
//
// Every swimbot is born with the INNATE weights plus its family's mutations.
// The innate weights reproduce the classic Gene Pool behavior (hungry -> eat,
// fed -> mate, threat -> flee), so the first generations survive; evolution
// then changes the behavior from there.
//
//   inputs  ──────────────────────────────►  outputs   (direct connections)
//      └──►  hidden (remembers its last state) ──┘
//------------------------------------------------------------------------------

// weight layout in the brain chromosome

const W_IO = 0;                                     // [core output][input]

const LEGACY_NUM_INPUTS = 21;

const LEGACY_W_HH = LEGACY_W_IH + NUM_HIDDEN * LEGACY_NUM_INPUTS;
const LEGACY_W_HO = LEGACY_W_HH + NUM_HIDDEN * NUM_HIDDEN;
export const LEGACY_BRAIN_SIZE = LEGACY_W_HO + LEGACY_NUM_OUTPUTS * NUM_HIDDEN;
const PRE_C12_W_IH = CORE_NUM_OUTPUTS * LEGACY_NUM_INPUTS;
const PRE_C12_W_HH = PRE_C12_W_IH + NUM_HIDDEN * LEGACY_NUM_INPUTS;
const PRE_C12_W_HO = PRE_C12_W_HH + NUM_HIDDEN * NUM_HIDDEN;
const PRE_C12_BRAIN_SIZE = PRE_C12_W_HO + CORE_NUM_OUTPUTS * NUM_HIDDEN;

function copySavedNetwork( target: Float32Array, source: ArrayLike<number>, outputs: number,
                           inputs: number, wIH: number, wHH: number, wHO: number ): void
{
    for ( let o = 0; o < outputs; o++ )
    {
        for ( let i = 0; i < inputs; i++ )
            target[ W_IO + o * NUM_INPUTS + i ] = Number.isFinite( Number( source[ o * inputs + i ] )) ? Number( source[ o * inputs + i ] ) : 0;
        for ( let h = 0; h < NUM_HIDDEN; h++ )
            target[ W_HO + o * NUM_HIDDEN + h ] = Number.isFinite( Number( source[ wHO + o * NUM_HIDDEN + h ] )) ? Number( source[ wHO + o * NUM_HIDDEN + h ] ) : 0;
    }
    for ( let h = 0; h < NUM_HIDDEN; h++ )
    {
        for ( let i = 0; i < inputs; i++ )
            target[ W_IH + h * NUM_INPUTS + i ] = Number.isFinite( Number( source[ wIH + h * inputs + i ] )) ? Number( source[ wIH + h * inputs + i ] ) : 0;
        for ( let j = 0; j < NUM_HIDDEN; j++ )
            target[ W_HH + h * NUM_HIDDEN + j ] = Number.isFinite( Number( source[ wHH + h * NUM_HIDDEN + j ] )) ? Number( source[ wHH + h * NUM_HIDDEN + j ] ) : 0;
    }
}

export function migrateLegacyBrainWeights( target: Float32Array, source: ArrayLike<number> ): boolean
{
    if ( source == null ) return false;
    if ( source.length === BRAIN_SIZE ) { target.set( source ); return true; }
    target.set( innateBrain() );
    if ( source.length === LEGACY_BRAIN_SIZE )
        copySavedNetwork( target, source, LEGACY_NUM_OUTPUTS, LEGACY_NUM_INPUTS, LEGACY_W_IH, LEGACY_W_HH, LEGACY_W_HO );
    else if ( source.length === PRE_C12_BRAIN_SIZE )
        copySavedNetwork( target, source, CORE_NUM_OUTPUTS, LEGACY_NUM_INPUTS, PRE_C12_W_IH, PRE_C12_W_HH, PRE_C12_W_HO );
    else return false;
    return true;
}

export const WEIGHT_LIMIT           = 8;
const BRAIN_MUTATION_RATE           = 0.03;   // per weight per birth
const BRAIN_MUTATION_STEP           = 0.25;
export const FOUNDER_BRAIN_NOISE    = 0.15;   // founders differ a little from the innate brain

export const THINK_PERIOD           = 5;      // steps between thoughts (spread over the swimbots)
export const DRIVE_HYSTERESIS       = 0.3;    // the current goal gets this bonus (no dithering)
export const EFFORT_MIN             = 0.5;
export const EFFORT_RANGE           = 0.7;    // effort = 0.5 .. 1.2
export const STEER_MAX_DEGREES      = 60;     // the largest turn the steering output can ask for

//------------------------------------------------------------
// the innate brain = the classic Gene Pool behavior
//------------------------------------------------------------
let _innate: Float32Array | null = null;

export function innateBrain(): Float32Array
{
    if ( _innate === null )
    {
        const w = new Float32Array( BRAIN_SIZE );
        const io = ( o, i, v ) => { w[ W_IO + o * NUM_INPUTS + i ] = v; };

        io( OUT_FOOD,    IN_HUNGER,  4 );      // hungry -> eat
        io( OUT_MATE,    IN_HUNGER, -4 );      // fed    -> mate
        io( OUT_HUNT,    IN_HUNGER,  4 );      // hungry predator -> hunt
        io( OUT_FLEE,    IN_THREAT, 10 );      // a hunting predator: flee when it comes within ~2/3 of the view
        io( OUT_FLEE,    IN_PREDATOR_NEAR, 4 ); // any predator very close: flee too
        io( OUT_FLEE,    IN_BIAS,   -3 );
        io( OUT_CLIMATE, IN_BIAS,    2 );      // follow the temperature more when it doesn't suit me
        io( OUT_CLIMATE, IN_FIT,    -3 );
        io( OUT_EFFORT,  IN_BIAS,    0.92 );   // effort 1.0 = classic swimming

        // hidden neurons start with tiny weights: unused, but ready for evolution
        _innate = w;
    }
    return _innate;
}

// a founder: innate brain with a little individual variation
export function founderBrain( out: Float32Array, includeExtended = false ): void
{
    const innate = innateBrain();
    out.set( innate );
    forEachLegacyWeightIndex( index => { out[index] = clampWeight( innate[index] + gaussian() * FOUNDER_BRAIN_NOISE ); });
    if ( includeExtended )
    {
        // No innate trigger: founders get a broader random response pattern,
        // then inheritance and selection can shape when the action is useful.
        const extendedFounderNoise = FOUNDER_BRAIN_NOISE;
        for ( let i = 0; i < LEGACY_NUM_INPUTS; i++ )
        {
            const index = W_IO + OUT_SELF_EAT * NUM_INPUTS + i;
            out[index] = clampWeight( gaussian() * extendedFounderNoise );
        }
        for ( let h = 0; h < NUM_HIDDEN; h++ )
        {
            const index = W_HO + OUT_SELF_EAT * NUM_HIDDEN + h;
            out[index] = clampWeight( gaussian() * extendedFounderNoise );
        }
    }
}

// Seed the new target-selection connections from the founder's existing
// identity without consuming the shared simulation RNG. That keeps BASE and
// C12 worlds identical at creation time while giving C12 founders variation.
export function seedPredatorChoiceBrain( w: Float32Array, seed: number ): void
{
    let state = ( seed >>> 0 ) || 1;
    const sample = () =>
    {
        state = ( Math.imul( state ^ ( state >>> 15 ), 0x2c1b3c6d ) + 0x297a2d39 ) >>> 0;
        state ^= state >>> 12;
        return ( state >>> 0 ) / 4294967296;
    };
    const noise = () => ( sample() * 2 - 1 ) * FOUNDER_BRAIN_NOISE;

    for ( let o = OUT_PREY_VALUE; o < NUM_OUTPUTS; o++ )
    {
        const io = extraOutputInputIndex( o );
        const ho = extraOutputHiddenIndex( o );
        for ( let i = 0; i < NUM_INPUTS; i++ ) w[ io + i ] = clampWeight( noise() );
        for ( let h = 0; h < NUM_HIDDEN; h++ ) w[ ho + h ] = clampWeight( noise() );
    }
}

//------------------------------------------------------------
// inheritance: each neuron (all its incoming weights) comes
// from one parent, then a few weights mutate
//------------------------------------------------------------
export function inheritBrain( child: Float32Array, a: Float32Array, b: Float32Array,
                              includeExtended = false, includePredatorChoice = false ): void
{
    child.set( innateBrain() );
    for ( let o = 0; o < LEGACY_NUM_OUTPUTS; o++ )
    {
        const p = ( gpRandom() < 0.5 ) ? a : b;
        copyRange( child, p, W_IO + o * NUM_INPUTS, NUM_INPUTS );
        copyRange( child, p, W_HO + o * NUM_HIDDEN, NUM_HIDDEN );
    }
    for ( let h = 0; h < NUM_HIDDEN; h++ )
    {
        const p = ( gpRandom() < 0.5 ) ? a : b;
        copyRange( child, p, W_IH + h * NUM_INPUTS, NUM_INPUTS );
        copyRange( child, p, W_HH + h * NUM_HIDDEN, NUM_HIDDEN );
    }
    if ( includeExtended )
    {
        const p = ( gpRandom() < 0.5 ) ? a : b;
        copyRange( child, p, W_IO + OUT_SELF_EAT * NUM_INPUTS, LEGACY_NUM_INPUTS );
        copyRange( child, p, W_HO + OUT_SELF_EAT * NUM_HIDDEN, NUM_HIDDEN );
    }
    if ( includePredatorChoice )
    {
        for ( let o = OUT_PREY_VALUE; o < NUM_OUTPUTS; o++ )
        {
            const p = ( gpRandom() < 0.5 ) ? a : b;
            copyRange( child, p, extraOutputInputIndex( o ), NUM_INPUTS );
            copyRange( child, p, extraOutputHiddenIndex( o ), NUM_HIDDEN );
        }
    }
    mutateBrain( child, BRAIN_MUTATION_RATE, includeExtended, includePredatorChoice );
}

export function mutateBrain( w: Float32Array, rate: number, includeExtended = false,
                             includePredatorChoice = false ): void
{
    const mutate = index => { if ( gpRandom() < rate ) w[index] = clampWeight( w[index] + gaussian() * BRAIN_MUTATION_STEP ); };
    forEachLegacyWeightIndex( mutate );
    if ( includeExtended )
    {
        for ( let i = 0; i < LEGACY_NUM_INPUTS; i++ )
        {
            const index = W_IO + OUT_SELF_EAT * NUM_INPUTS + i;
            mutate( index );
        }
        for ( let h = 0; h < NUM_HIDDEN; h++ )
        {
            const index = W_HO + OUT_SELF_EAT * NUM_HIDDEN + h;
            mutate( index );
        }
    }
    if ( includePredatorChoice )
    {
        for ( let o = OUT_PREY_VALUE; o < NUM_OUTPUTS; o++ )
        {
            const io = extraOutputInputIndex( o );
            const ho = extraOutputHiddenIndex( o );
            for ( let i = 0; i < NUM_INPUTS; i++ ) mutate( io + i );
            for ( let h = 0; h < NUM_HIDDEN; h++ ) mutate( ho + h );
        }
    }
}

function forEachLegacyWeightIndex( visit: ( index: number ) => void ): void
{
    for ( let o = 0; o < LEGACY_NUM_OUTPUTS; o++ )
        for ( let i = 0; i < LEGACY_NUM_INPUTS; i++ ) visit( W_IO + o * NUM_INPUTS + i );
    for ( let h = 0; h < NUM_HIDDEN; h++ )
        for ( let i = 0; i < LEGACY_NUM_INPUTS; i++ ) visit( W_IH + h * NUM_INPUTS + i );
    for ( let h = 0; h < NUM_HIDDEN; h++ )
        for ( let j = 0; j < NUM_HIDDEN; j++ ) visit( W_HH + h * NUM_HIDDEN + j );
    for ( let o = 0; o < LEGACY_NUM_OUTPUTS; o++ )
        for ( let h = 0; h < NUM_HIDDEN; h++ ) visit( W_HO + o * NUM_HIDDEN + h );
}

function extraOutputInputIndex( output: number ): number
{
    return EXTRA_OUTPUT_START + ( output - CORE_NUM_OUTPUTS ) * ( NUM_INPUTS + NUM_HIDDEN );
}

function extraOutputHiddenIndex( output: number ): number
{
    return extraOutputInputIndex( output ) + NUM_INPUTS;
}

function copyRange( to: Float32Array, from: Float32Array, start: number, length: number ): void
{
    for ( let k = start; k < start + length; k++ ) { to[k] = from[k]; }
}

function clampWeight( v: number ): number
{
    return v > WEIGHT_LIMIT ? WEIGHT_LIMIT : ( v < -WEIGHT_LIMIT ? -WEIGHT_LIMIT : v );
}

//------------------------------------------------------------
// one thought: inputs -> outputs (hidden keeps its state)
//------------------------------------------------------------
export function think( w: Float32Array, inputs: Float64Array, hidden: Float64Array, newHidden: Float64Array, outputs: Float64Array ): void
{
    for ( let h = 0; h < NUM_HIDDEN; h++ )
    {
        let sum = 0;
        const ih = W_IH + h * NUM_INPUTS;
        for ( let i = 0; i < NUM_INPUTS; i++ ) { sum += w[ ih + i ] * inputs[i]; }
        const hh = W_HH + h * NUM_HIDDEN;
        for ( let j = 0; j < NUM_HIDDEN; j++ ) { sum += w[ hh + j ] * hidden[j]; }
        newHidden[h] = Math.tanh( sum );
    }
    for ( let h = 0; h < NUM_HIDDEN; h++ ) { hidden[h] = newHidden[h]; }

    for ( let o = 0; o < NUM_OUTPUTS; o++ )
    {
        let sum = 0;
        const io = ( o < CORE_NUM_OUTPUTS ) ? W_IO + o * NUM_INPUTS : extraOutputInputIndex( o );
        for ( let i = 0; i < NUM_INPUTS; i++ ) { sum += w[ io + i ] * inputs[i]; }
        const ho = ( o < CORE_NUM_OUTPUTS ) ? W_HO + o * NUM_HIDDEN : extraOutputHiddenIndex( o );
        for ( let h = 0; h < NUM_HIDDEN; h++ ) { sum += w[ ho + h ] * hidden[h]; }
        outputs[o] = sum;
    }
}

export function sigmoid( x: number ): number { return 1 / ( 1 + Math.exp( -x )); }

// how far a brain has moved away from the innate one (mean absolute difference)
export function brainDistanceFromInnate( w: Float32Array, includeExtended = false,
                                         includePredatorChoice = false ): number
{
    const innate = innateBrain();
    let sum = 0, count = 0;
    forEachLegacyWeightIndex( index => { sum += Math.abs( w[index] - innate[index] ); count ++; });
    if ( includeExtended )
    {
        for ( let i = 0; i < LEGACY_NUM_INPUTS; i++ )
        {
            const index = W_IO + OUT_SELF_EAT * NUM_INPUTS + i;
            sum += Math.abs( w[index] - innate[index] ); count ++;
        }
        for ( let h = 0; h < NUM_HIDDEN; h++ )
        {
            const index = W_HO + OUT_SELF_EAT * NUM_HIDDEN + h;
            sum += Math.abs( w[index] - innate[index] ); count ++;
        }
    }
    if ( includePredatorChoice )
    {
        for ( let o = OUT_PREY_VALUE; o < NUM_OUTPUTS; o++ )
        {
            const io = extraOutputInputIndex( o ), ho = extraOutputHiddenIndex( o );
            for ( let i = 0; i < NUM_INPUTS; i++ ) { sum += Math.abs( w[ io + i ] - innate[ io + i ] ); count ++; }
            for ( let h = 0; h < NUM_HIDDEN; h++ ) { sum += Math.abs( w[ ho + h ] - innate[ ho + h ] ); count ++; }
        }
    }
    return count > 0 ? sum / count : 0;
}

//------------------------------------------------------------------------------
// the brain's state machine: one row of the mind (see MindLayout.ts).
// Used by the Brain below and by the decision kernel (any thread).
// rand() gives a random number 0..1.
//------------------------------------------------------------------------------
export function brainUpdate( M: Float64Array, m: number, rand: () => number ): void
{
    //----------------------------------------------------------------------
    // neural brain: the network has chosen the goal (see NeuralBrain.ts);
    // the looking / pursuing steps below stay the same.
    // Classic brain: fleeing a predator in reach comes first (herbivores),
    // then food (prey for predators) when hungry, otherwise a mate.
    //----------------------------------------------------------------------
    let state = M[ m + ML.M_STATE ];
    const goal = M[ m + ML.M_GOAL ];

    if ( goal >= 0 )
    {
        if ( goal === OUT_SELF_EAT ) { M[ m + ML.M_STATE ] = BRAIN_STATE_SELF_FEED;        return; }
        if ( goal === OUT_FLEE     ) { M[ m + ML.M_STATE ] = BRAIN_STATE_FLEEING_PREDATOR; return; }
        if (( goal === OUT_HUNT ) && ( state != BRAIN_STATE_PURSUING_PREY ) && ( state != BRAIN_STATE_LOOKING_FOR_PREY )) { state = BRAIN_STATE_LOOKING_FOR_PREY; }
        if (( goal === OUT_FOOD ) && ( state != BRAIN_STATE_PURSUING_FOOD ) && ( state != BRAIN_STATE_LOOKING_FOR_FOOD )) { state = BRAIN_STATE_LOOKING_FOR_FOOD; }
        if (( goal === OUT_MATE ) && ( state != BRAIN_STATE_PURSUING_MATE ) && ( state != BRAIN_STATE_LOOKING_FOR_MATE )) { state = BRAIN_STATE_LOOKING_FOR_MATE; }
    }
    else if (( M[ m + ML.M_THREAT_NEAR ] === 1 ) && ( M[ m + ML.M_PRED_LEVEL ] <= PREDATOR_LEVEL_THRESHOLD ))
    {
        M[ m + ML.M_STATE ] = BRAIN_STATE_FLEEING_PREDATOR;
        return;
    }
    else if ( M[ m + ML.M_BRAIN_ENERGY ] < M[ m + ML.M_HUNGER ] )
    {
        if ( M[ m + ML.M_PRED_LEVEL ] > PREDATOR_LEVEL_THRESHOLD )
        {
            if (( state != BRAIN_STATE_PURSUING_PREY ) && ( state != BRAIN_STATE_LOOKING_FOR_PREY )) { state = BRAIN_STATE_LOOKING_FOR_PREY; }
        }
        else
        {
            if (( state != BRAIN_STATE_PURSUING_FOOD ) && ( state != BRAIN_STATE_LOOKING_FOR_FOOD )) { state = BRAIN_STATE_LOOKING_FOR_FOOD; }
        }
    }
    else
    {
        if (( state != BRAIN_STATE_PURSUING_MATE ) && ( state != BRAIN_STATE_LOOKING_FOR_MATE )) { state = BRAIN_STATE_LOOKING_FOR_MATE; }
    }

    const foundFood = M[ m + ML.M_FOUND_FOOD ] === 1;
    const foundMate = M[ m + ML.M_FOUND_MATE ] === 1;
    const foundPrey = M[ m + ML.M_FOUND_PREY ] === 1;

    if      ( state == BRAIN_STATE_LOOKING_FOR_FOOD ) { if (  foundFood ) state = BRAIN_STATE_PURSUING_FOOD;    }
    else if ( state == BRAIN_STATE_PURSUING_FOOD    ) { if ( !foundFood ) state = BRAIN_STATE_LOOKING_FOR_FOOD; }
    else if ( state == BRAIN_STATE_LOOKING_FOR_PREY ) { if (  foundPrey ) state = BRAIN_STATE_PURSUING_PREY;    }
    else if ( state == BRAIN_STATE_PURSUING_PREY    )
    {
        // no switching at once (it would dither): a small chance per step
        if ( !foundPrey && ( rand() < 0.1 )) state = BRAIN_STATE_LOOKING_FOR_PREY;
    }
    else if ( state == BRAIN_STATE_LOOKING_FOR_MATE ) { if (  foundMate ) state = BRAIN_STATE_PURSUING_MATE;    }
    else if ( state == BRAIN_STATE_PURSUING_MATE    ) { if ( !foundMate ) state = BRAIN_STATE_LOOKING_FOR_MATE; }

    M[ m + ML.M_STATE ] = state;
}

//----------------------------------------
// Brain! (its numbers live in a mind row:
// a private one until bind() gives it
// the swimbot's row in the shared state)
//----------------------------------------
export function Brain()
{
    let _M = new Float64Array( ML.MIND_STRIDE );
    let _m = 0;
    _M[ ML.M_STATE ]      = BRAIN_STATE_NULL;
    _M[ ML.M_GOAL ]       = -1;
    _M[ ML.M_ATTRACTION ] = ATTRACTION_SIMILAR_COLOR;

    // use the row m of the array M from now on (the numbers move along)
    this.bind = function( M: Float64Array, m: number )
    {
        M.set( _M.subarray( _m, _m + ML.MIND_STRIDE ), m );
        _M = M;
        _m = m;
    }

    //-----------------------------
    this.initialize = function()
    {
        _M[ _m + ML.M_STATE ]       = BRAIN_STATE_NULL;
        _M[ _m + ML.M_PRED_LEVEL ]  = ZERO;
        _M[ _m + ML.M_FOUND_PREY ]  = 0;
        _M[ _m + ML.M_THREAT_NEAR ] = 0;
        _M[ _m + ML.M_GOAL ]        = -1;
    }

    this.update = function() { brainUpdate( _M, _m, gpRandom ); }

    //-------------------------------------------------------------------
    // setters
    //-------------------------------------------------------------------
    this.setEnergyLevel     = function(e) { _M[ _m + ML.M_BRAIN_ENERGY ] = e; }
    this.setHungerThreshold = function(h) { _M[ _m + ML.M_HUNGER ]       = h; }
    this.setFoundFoodBit    = function(f) { _M[ _m + ML.M_FOUND_FOOD ]   = f ? 1 : 0; }
    this.setFoundSwimbot    = function(f) { _M[ _m + ML.M_FOUND_MATE ]   = f ? 1 : 0; }
    this.setFoundPrey       = function(f) { _M[ _m + ML.M_FOUND_PREY ]   = f ? 1 : 0; }
    this.setPredatorLevel        = function(level) { _M[ _m + ML.M_PRED_LEVEL ]  = level; }
    this.setNearbyPredatorThreat = function(v)     { _M[ _m + ML.M_THREAT_NEAR ] = v ? 1 : 0; }
    this.returnToSearchingForPrey = function() { if ( _M[ _m + ML.M_STATE ] === BRAIN_STATE_PURSUING_PREY ) _M[ _m + ML.M_STATE ] = BRAIN_STATE_LOOKING_FOR_PREY; }
    this.setGoal                 = function(g)     { _M[ _m + ML.M_GOAL ] = g; }
    this.getGoal                 = function()      { return _M[ _m + ML.M_GOAL ]; }

    //--------------------------------
    this.setAttraction = function(a)
    {
        _M[ _m + ML.M_ATTRACTION ] = a;

        // forgetting the found swimbot makes the swimbot search for a new potential mate
        _M[ _m + ML.M_FOUND_MATE ] = 0;
    }

    //--------------------------------------------------------------------------
    // getters
    //--------------------------------------------------------------------------
    this.getHungerThreshold     = function() { return _M[ _m + ML.M_HUNGER ];     }
    this.getAttractionCriterion = function() { return _M[ _m + ML.M_ATTRACTION ]; }
    this.getState               = function() { return _M[ _m + ML.M_STATE ];      }
    this.getPredatorLevel       = function() { return _M[ _m + ML.M_PRED_LEVEL ]; }
}

import { simClock, gpRandom } from "./Globals";
import { NULL_INDEX, ONE, ONE_HALF, PI2, ZERO } from "./MathConstants";
import { SPAWN_FOOD_RANDOMLY_IN_POOL } from "./Parameters";
import { POOL_BOTTOM, POOL_HEIGHT, POOL_LEFT, POOL_RIGHT, POOL_TOP, POOL_WIDTH } from "./Pool";
import { assert } from "./Utility";
import { SplitVector2D, Vector2D } from "./Vector2D";

export const FOOD_BIT_SIZE                     = 1.5; //default size
export const MIN_FOOD_BIT_MAX_SPAWN_RADIUS     = 10;
export const MAX_FOOD_BIT_MAX_SPAWN_RADIUS     = 4000.0;
export const DEFAULT_FOOD_BIT_MAX_SPAWN_RADIUS = 4000.0; //max distance for spawned child
export const MIN_FOOD_BIT_ENERGY               = 0.0;
export const MAX_FOOD_BIT_ENERGY               = 100.0;
export const DEFAULT_FOOD_BIT_ENERGY           = 50.0;  //when eaten, swimbot gets this much energy
export const FOOD_BIT_SIZE_VIEW_SCALE          = 0.03; //increase with view scale (a kind of LOD)
export const FOOD_BIT_GRAB_RADIUS              = 20.0;  // radius for grabbing food bit
export const FOOD_BIT_BOUNDARY_MARGIN          = POOL_WIDTH * 0.01; // important value - creates empty space from wall
export const FOOD_BIT_COLOR_COMPONENTS         = "100, 200, 100";
export const FOOD_BIT_ROLLOVER_COLOR           = "rgba( 100, 200, 100, 0.5 )";
export const FOOD_BIT_SELECT_COLOR             = "rgba( 200, 200, 200, 1.0 )";
export const FOOD_OPACITY_INCREMENT            = 0.01;


//------------------------
// Food bit
//------------------------
export function FoodBit()
{
    // position, type and energy live in the shared state once the food bit
    // has its slot (see bindShared), so the threads see them without copying
    let _fx = new Float64Array( 1 ), _fy = new Float64Array( 1 ), _fi = 0;
    let _fe = new Float64Array( 1 );
    let _ft = new Uint8Array( 1 );
    const _position     = new SplitVector2D( _fx, _fy, 0 );
    let _red            = ZERO;
    let _green          = ZERO;
    let _blue           = ZERO;
    let _index          = NULL_INDEX;
    let _fb = new Float64Array( 1 );   // the step it appeared (shared: foodBorn)
    let _maxSpawnRadius = DEFAULT_FOOD_BIT_MAX_SPAWN_RADIUS;
    let _autotomyDecoy  = false;
    let _shownFully     = false;   // (held in the "hand": no fading in)

    // alive or not, mirrored as one byte in the shared state (see bindLiveFlag)
    let _live   = new Uint8Array( 1 );
    let _liveAt = 0;
    const markLive = function() { _live[ _liveAt ] = ( _index != NULL_INDEX ) ? 1 : 0; }

    this.bindShared = function( st, slot: number )
    {
        _live   = st.liveFood;
        _liveAt = slot;
        markLive();
        st.foodEnergy[ slot ] = _fe[ _fi ];
        st.foodType  [ slot ] = _ft[ _fi ];
        st.foodBorn  [ slot ] = _fb[ _fi ];
        _fe = st.foodEnergy;
        _fb = st.foodBorn;
        _ft = st.foodType;
        _fi = slot;
        _position.rebind( st.foodX, st.foodY, slot );
    }

    //--------------------------------------------------------
    // initialize
    //--------------------------------------------------------
	this.initialize = function(f)
	{
        _index      = f;
        _shownFully = false;
        _fb[ _fi ] = simClock.step;
        _fe[ _fi ]     = DEFAULT_FOOD_BIT_ENERGY;
        _autotomyDecoy = false;
        _position.x = POOL_LEFT + gpRandom() * POOL_WIDTH;
        _position.y = POOL_TOP  + gpRandom() * POOL_HEIGHT;
        _ft[ _fi ]       = 0;
        markLive();
    }



    //--------------------------------------
	this.setType = function(n)
	{
//assert ( ( n === 0 ) || ( n === 1 ), "Foodbit.js: setType: ( n === 0 ) || ( n === 1 )" );

	    _ft[ _fi ] = n;

        this.setColorAccordingToType();
	}

    this.setAutotomyDecoy = function( value ) { _autotomyDecoy = value === true; }




    //------------------------------------------------
	this.setColorAccordingToType = function()
	{
       if ( _ft[ _fi ] === 0 ) { _red = 0.3; _green = 0.8; _blue = 0.2; }
       if ( _ft[ _fi ] === 1 ) { _red = 0.3; _green = 0.5; _blue = 0.9; }
       if ( _ft[ _fi ] === 2 ) { _red = 0.9; _green = 0.1; _blue = 0.1; }
	   }



    //----------------------------------------------------------------------------
	this.spawnFromParent = function( parentFoodBit, childIndex, childType )
	{
	       assert( parentFoodBit.getIndex() != NULL_INDEX, "foodbit.js: spawnNearParent: parentFoodBit.index != NULL_INDEX" );
	       assert( parentFoodBit.getAlive(), "foodbit.js: spawnNearParent: parentFoodBit.getAlive()" );
	       assert( childIndex != NULL_INDEX, "foodbit.js: spawnNearParent: childIndex != NULL_INDEX" );

        _index      = childIndex;
        markLive();
        _fb[ _fi ] = simClock.step;
        _fe[ _fi ]     = parentFoodBit.getEnergy();
        _ft[ _fi ]       = childType;
        _autotomyDecoy = false;

        this.setColorAccordingToType();

        //-----------------------------
        // set the position
        //-----------------------------
        _position.set( parentFoodBit.getPosition() );

              //-----------------------------
              // randomize position
              //-----------------------------
              this.randomizeSpawnPosition( parentFoodBit );

    }

    // GenePool.updateFood calls it by this name
    this.setSpawnPositionRelativeToParent = this.spawnFromParent;




    //-----------------------------------------------------------
	this.randomizeSpawnPosition = function( parentFoodBit )
    {
        _position.set( parentFoodBit.getPosition() );

        let xx = gpRandom() * gpRandom();
        let yy = gpRandom() * gpRandom();

        if ( gpRandom() < ONE_HALF ) { xx *= -ONE; }
        if ( gpRandom() < ONE_HALF ) { yy *= -ONE; }

        _position.x += xx * _maxSpawnRadius;
        _position.y += yy * _maxSpawnRadius;

        //-----------------------------
        // pool boundary collisions
        //-----------------------------
        let pb = POOL_TOP       + FOOD_BIT_BOUNDARY_MARGIN;
        let pt = POOL_BOTTOM    - FOOD_BIT_BOUNDARY_MARGIN;
        let pl = POOL_LEFT	    + FOOD_BIT_BOUNDARY_MARGIN;
        let pr = POOL_RIGHT	    - FOOD_BIT_BOUNDARY_MARGIN;

        //console.log( "before:" + _position.y );

                if ( _position.y < pb ) { _position.y += ( ( pb - _position.y ) * 2 ); }
        else	if ( _position.y > pt ) { _position.y += ( ( pt - _position.y ) * 2 ); }
                if ( _position.x > pr ) { _position.x += ( ( pr - _position.x ) * 2 ); }
        else	if ( _position.x < pl ) { _position.x += ( ( pl - _position.x ) * 2 ); }

        //console.log( "after:" + _position.y );

        if ( SPAWN_FOOD_RANDOMLY_IN_POOL )
        {
            _position.x = POOL_LEFT + gpRandom() * POOL_WIDTH;
            _position.y = POOL_TOP  + gpRandom() * POOL_HEIGHT;
        }


        assert( _position.x < POOL_RIGHT,   "foodbit.js: spawnNearParent: _position.x < POOL_RIGHT"  );
        assert( _position.x > POOL_LEFT,    "foodbit.js: spawnNearParent: _position.x > POOL_LEFT"   );
        assert( _position.y > POOL_TOP,     "foodbit.js: spawnNearParent: _position.y < POOL_TOP"	);
        assert( _position.y < POOL_BOTTOM,  "foodbit.js: spawnNearParent: _position.y > POOL_BOTTOM" );
    }



    //-----------------------------
	this.setPosition = function(p)
	{
        _position.set(p);

                if ( _position.y < POOL_TOP    ) { _position.y = POOL_TOP    + FOOD_BIT_SIZE; }
        else	if ( _position.y > POOL_BOTTOM ) { _position.y = POOL_BOTTOM - FOOD_BIT_SIZE; }
                if ( _position.x > POOL_RIGHT  ) { _position.x = POOL_RIGHT  - FOOD_BIT_SIZE; }
        else	if ( _position.x < POOL_LEFT   ) { _position.x = POOL_LEFT   + FOOD_BIT_SIZE; }

        assert( _position.x < POOL_RIGHT,   "foodbit.js: setPosition: _position.x < POOL_RIGHT"  );
        assert( _position.x > POOL_LEFT,    "foodbit.js: setPosition: _position.x > POOL_LEFT"   );
        assert( _position.y > POOL_TOP,     "foodbit.js: setPosition: _position.y < POOL_TOP"	);
        assert( _position.y < POOL_BOTTOM,  "foodbit.js: setPosition: _position.y > POOL_BOTTOM" );
    }



    //-------------------------------
	this.shiftPosition = function(s)
	{
	    _position.x += s.x;
	    _position.y += s.y;
    }

    //-----------------------------------
	this.setMaxSpawnRadius = function(r)
	{
	    _maxSpawnRadius = r;

	    assert( _maxSpawnRadius <= MAX_FOOD_BIT_MAX_SPAWN_RADIUS, "FoodBit: setMaxSpawnRadius: _maxSpawnRadius <= MAX_FOOD_BIT_MAX_SPAWN_RADIUS" );

	    assert( _maxSpawnRadius >= MIN_FOOD_BIT_MAX_SPAWN_RADIUS, "FoodBit: setMaxSpawnRadius: _maxSpawnRadius >= MIN_FOOD_BIT_MAX_SPAWN_RADIUS" );
    }


    //----------------------------------
	this.getMaxSpawnRadius = function()
	{
	    return _maxSpawnRadius;
    }

    //----------------------------
	this.setEnergy = function(e)
	{
	    _fe[ _fi ] = e;

	    assert( _fe[ _fi ] <= MAX_FOOD_BIT_ENERGY, "FoodBit:getMaxSpawnRadius setEnergy: _fe[ _fi ] <= MAX_FOOD_BIT_ENERGY" );
	    assert( _fe[ _fi ] >= MIN_FOOD_BIT_ENERGY, "FoodBit:getMaxSpawnRadius setEnergy: _fe[ _fi ] >= MIN_FOOD_BIT_ENERGY" );
	   }

    //----------------------
    this.kill = function()
    {
        _index = NULL_INDEX;
        markLive();
    }

	//----------------------------
	// getters
	//----------------------------
	this.getPosition    = function() { return _position;    }
	this.getEnergy      = function() { return _fe[ _fi ];      }
	this.getType   = function() { return _ft[ _fi ];   }
	this.getAutotomyDecoy = function() { return _autotomyDecoy; }
	this.getIndex       = function() { return _index;       }
	this.getAge         = function() { return simClock.step - _fb[ _fi ]; }

	// a new food bit fades in
	const opacity = function() { return _shownFully ? ONE : Math.min( ONE, ( simClock.step - _fb[ _fi ] ) * FOOD_OPACITY_INCREMENT ); }
	this.showFully = function() { _shownFully = true; }
    this.getAlive       = function() { return ( _index != NULL_INDEX ); }

    // Add method for spatial grid support
    this.getBoundingRadius = function() { return FOOD_BIT_GRAB_RADIUS; }

	// (nothing to do every step: the age and the fading in follow from the clock)
	this.update = function() {}

	//--------------------------------
	// render
	//--------------------------------
	//----------------------------------------------------
	// hand this food bit to the graphics-card renderer
	//----------------------------------------------------
	this.writeCapsule = function( capsules, viewScale )
	{
	    let radius = FOOD_BIT_SIZE + viewScale * FOOD_BIT_SIZE_VIEW_SCALE * FOOD_BIT_SIZE_VIEW_SCALE;
	    capsules.add( _position.x, _position.y, _position.x, _position.y, radius, _red, _green, _blue, opacity() );
	}

	this.render = function( vewScale )
	{
	    canvas.fillStyle
        = "rgba( "
        + Math.floor( _red   * 255 ) + ", "
        + Math.floor( _green * 255 ) + ", "
        + Math.floor( _blue  * 255 ) + ", "
        + opacity() + ")";

	    let radius = FOOD_BIT_SIZE + vewScale * FOOD_BIT_SIZE_VIEW_SCALE * FOOD_BIT_SIZE_VIEW_SCALE;

        canvas.beginPath();
        canvas.arc( _position.x, _position.y, radius, 0, PI2, false );
        canvas.fill();
        canvas.closePath();
    }

	//----------------------------
	// render moused-over outline
	//---------------------------
	this.renderMousedOverOutline = function( viewScale )
	{
	    this.showSelectCircle( viewScale, FOOD_BIT_ROLLOVER_COLOR );
    }

	//----------------------------
	// render select outline
	//---------------------------
	this.renderSelectOutline = function( viewScale )
	{
	    this.showSelectCircle( viewScale, FOOD_BIT_SELECT_COLOR );
    }

	//-----------------------------------------------------
	this.showSelectCircle = function( viewScale, color )
    {
	    let lineWidth = 1.0 + 0.005 * viewScale;

        canvas.lineWidth = lineWidth;
        canvas.strokeStyle = "rgba( 100, 200, 100, 0.05 )";
        canvas.beginPath();
        canvas.arc( _position.x, _position.y, FOOD_BIT_GRAB_RADIUS, 0, PI2, false );
        canvas.stroke();
        canvas.closePath();

        canvas.lineWidth = lineWidth * 0.3;
        canvas.strokeStyle = "rgba( 100, 200, 100, 0.1 )";
        canvas.beginPath();
        canvas.arc( _position.x, _position.y, FOOD_BIT_GRAB_RADIUS, 0, PI2, false );
        canvas.stroke();
        canvas.closePath();
     }

}
